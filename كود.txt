from google import genai

API_KEY = "ضع_مفتاحك_هنا"

print("جاري الاتصال بـ Gemini...", flush=True)

try:
    client = genai.Client(
        api_key=API_KEY,
        http_options={"timeout": 20000}
    )

    print("تم إعداد الاتصال.", flush=True)
    print("جارٍ إرسال السؤال...", flush=True)

    response = client.models.generate_content(
        model="gemini-3.8-flash",
        contents="Reply with only: CONNECTION_OK"
    )

    print("الرد:", response.text, flush=True)

except Exception as e:
    print("حدث خطأ:", repr(e), flush=True)
