<script>
    let currentConversationId = null;
    let isGenerating = false;
    let activeController = null;
    let stopRequested = false;

    const messagesElement = document.getElementById("messages");
    const inputElement = document.getElementById("messageInput");
    const sendButton = document.getElementById("sendButton");
    const stopButton = document.getElementById("stopButton");
    const statusElement = document.getElementById("status");


    // نسخ النص إلى الحافظة
    async function copyText(text, button) {
        try {
            if (navigator.clipboard && window.isSecureContext) {
                await navigator.clipboard.writeText(text);
            } else {
                // طريقة احتياطية للمتصفحات التي لا تدعم Clipboard API
                const temporary = document.createElement("textarea");

                temporary.value = text;
                temporary.style.position = "fixed";
                temporary.style.left = "-9999px";
                temporary.style.top = "0";

                document.body.appendChild(temporary);
                temporary.focus();
                temporary.select();

                const successful = document.execCommand("copy");

                temporary.remove();

                if (!successful) {
                    throw new Error("تعذر النسخ");
                }
            }

            const oldText = button.textContent;

            button.textContent = "✓ تم النسخ";
            button.classList.add("copied");

            setTimeout(() => {
                button.textContent = "📋 نسخ";
                button.classList.remove("copied");
            }, 1800);

        } catch (error) {
            button.textContent = "تعذر النسخ";

            setTimeout(() => {
                button.textContent = "📋 نسخ";
            }, 1800);
        }
    }


    // عرض النص وإنشاء زر نسخ لكل كتلة كود
    function renderMessageContent(element, content) {
        element.replaceChildren();

        const parts = String(content).split(/(```[\s\S]*?```)/g);

        parts.forEach(part => {

            if (part.startsWith("```") && part.endsWith("```")) {

                const inner = part.slice(3, -3);

                const firstNewline = inner.indexOf("\n");

                let language = "Code";
                let code = inner;

                if (firstNewline !== -1) {
                    const possibleLanguage =
                        inner.slice(0, firstNewline).trim();

                    if (
                        possibleLanguage &&
                        !possibleLanguage.includes(" ") &&
                        possibleLanguage.length < 30
                    ) {
                        language = possibleLanguage;
                        code = inner.slice(firstNewline + 1);
                    }
                }

                const container = document.createElement("div");
                container.className = "code-container";

                const header = document.createElement("div");
                header.className = "code-header";

                const languageLabel = document.createElement("span");
                languageLabel.className = "code-language";
                languageLabel.textContent = language;

                const copyButton = document.createElement("button");
                copyButton.className = "copy-code";
                copyButton.textContent = "📋 نسخ";

                copyButton.addEventListener("click", () => {
                    copyText(code, copyButton);
                });

                const pre = document.createElement("pre");
                const codeElement = document.createElement("code");

                // استخدام textContent لمنع تنفيذ الكود كـ HTML
                codeElement.textContent = code;

                header.appendChild(languageLabel);
                header.appendChild(copyButton);

                pre.appendChild(codeElement);

                container.appendChild(header);
                container.appendChild(pre);

                element.appendChild(container);

            } else if (part) {

                const textNode = document.createElement("span");
                textNode.className = "message-text";
                textNode.textContent = part;

                element.appendChild(textNode);
            }
        });

        messagesElement.scrollTop = messagesElement.scrollHeight;
    }


    // إضافة رسالة إلى الواجهة
    function addMessage(role, content, model = "") {
        const welcome = document.getElementById("welcome");

        if (welcome) {
            welcome.remove();
        }

        const box = document.createElement("div");
        box.className = "message " + role;

        const roleLabel = document.createElement("div");
        roleLabel.className = "role";

        roleLabel.textContent =
            role === "user"
                ? "أنت"
                : "المساعد" + (model ? " · " + model : "");

        const text = document.createElement("div");
        text.className = "message-content";

        box.appendChild(roleLabel);
        box.appendChild(text);

        messagesElement.appendChild(box);

        renderMessageContent(text, content);

        return box;
    }


    function showWelcome() {
        messagesElement.innerHTML = `
            <div class="welcome" id="welcome">
                <h1>كيف يمكنني مساعدتك؟</h1>
                <p>اكتب سؤالك للبدء.</p>
            </div>
        `;
    }


    // تحميل قائمة المحادثات
    async function refreshConversations() {
        try {
            const response = await fetch("/conversations");
            const conversations = await response.json();

            const list = document.getElementById("conversationList");
            list.innerHTML = "";

            conversations.forEach(conversation => {
                const item = document.createElement("div");

                item.className = "conversation";

                if (conversation.id === currentConversationId) {
                    item.classList.add("active");
                }

                item.textContent = conversation.title;
                item.title = conversation.title;

                item.onclick = () => loadConversation(conversation.id);

                list.appendChild(item);
            });

        } catch (error) {
            statusElement.textContent =
                "تعذر تحميل قائمة المحادثات.";
        }
    }


    // إنشاء محادثة جديدة
    async function newChat() {
        if (isGenerating) {
            statusElement.textContent =
                "أوقف الإجابة الحالية أولاً قبل إنشاء محادثة جديدة.";
            return;
        }

        try {
            const response = await fetch("/conversations", {
                method: "POST"
            });

            const data = await response.json();

            currentConversationId = data.id;

            showWelcome();

            statusElement.textContent = "";

            await refreshConversations();

            inputElement.focus();

        } catch (error) {
            statusElement.textContent =
                "تعذر إنشاء محادثة جديدة.";
        }
    }


    // تحميل محادثة محفوظة
    async function loadConversation(id) {
        if (isGenerating) {
            statusElement.textContent =
                "أوقف الإجابة الحالية أولاً.";
            return;
        }

        try {
            const response = await fetch(
                "/conversations/" + encodeURIComponent(id)
            );

            const data = await response.json();

            if (!response.ok) {
                throw new Error(
                    data.error || "فشل تحميل المحادثة"
                );
            }

            currentConversationId = data.id;

            messagesElement.innerHTML = "";

            if (!data.messages.length) {
                showWelcome();
            } else {
                data.messages.forEach(message => {
                    addMessage(
                        message.role === "assistant"
                            ? "assistant"
                            : "user",
                        message.content,
                        message.model || ""
                    );
                });
            }

            statusElement.textContent = "";

            await refreshConversations();

        } catch (error) {
            statusElement.textContent =
                "تعذر تحميل المحادثة: " + error.message;
        }
    }


    // إرسال الرسالة إلى Gemini
    async function sendMessage() {
        const message = inputElement.value.trim();

        if (!message || isGenerating) {
            return;
        }

        try {
            if (!currentConversationId) {
                const response = await fetch("/conversations", {
                    method: "POST"
                });

                const data = await response.json();

                currentConversationId = data.id;
            }

            const model = document.getElementById("model").value;

            addMessage("user", message);

            inputElement.value = "";

            isGenerating = true;
            stopRequested = false;

            activeController = new AbortController();

            sendButton.disabled = true;
            stopButton.style.display = "inline-block";

            statusElement.textContent = "جارٍ التفكير...";

            const pendingBox = addMessage(
                "assistant",
                "جارٍ إنشاء الإجابة..."
            );

            const response = await fetch("/chat", {
                method: "POST",

                headers: {
                    "Content-Type": "application/json"
                },

                signal: activeController.signal,

                body: JSON.stringify({
                    message: message,
                    model: model,
                    conversation_id: currentConversationId
                })
            });

            const data = await response.json();

            const contentElement =
                pendingBox.querySelector(".message-content");

            if (stopRequested) {
                renderMessageContent(
                    contentElement,
                    "تم إيقاف انتظار الإجابة. قد يستمر الطلب على الخادم."
                );

                return;
            }

            if (!response.ok || data.error) {
                renderMessageContent(
                    contentElement,
                    "حدث خطأ: " +
                    (data.error || "تعذر الحصول على الإجابة")
                );

                return;
            }

            currentConversationId = data.conversation_id;

            renderMessageContent(
                contentElement,
                data.reply || "لم يصل رد نصي."
            );

            pendingBox.querySelector(".role").textContent =
                "المساعد · " + (data.model || model);

            statusElement.textContent = "";

        } catch (error) {

            if (stopRequested || error.name === "AbortError") {
                statusElement.textContent =
                    "تم إيقاف انتظار الإجابة.";

            } else {
                statusElement.textContent =
                    "تعذر الاتصال: " + error.message;
            }

        } finally {

            isGenerating = false;
            activeController = null;

            sendButton.disabled = false;
            stopButton.style.display = "none";

            if (!stopRequested) {
                statusElement.textContent = "";
            }

            await refreshConversations();
        }
    }


    // إيقاف انتظار الإجابة
    async function stopGeneration() {
        if (!isGenerating) {
            return;
        }

        stopRequested = true;

        statusElement.textContent =
            "جارٍ إيقاف الانتظار...";

        if (activeController) {
            activeController.abort();
        }

        statusElement.textContent =
            "أُوقف انتظار الإجابة. قد تستمر المعالجة على خوادم المزود.";
    }


    // Enter للإرسال و Shift+Enter لسطر جديد
    function handleKey(event) {
        if (
            event.key === "Enter" &&
            !event.shiftKey &&
            !event.isComposing
        ) {
            event.preventDefault();
            sendMessage();
        }
    }


    // بدء تشغيل الواجهة
    async function initialize() {
        await refreshConversations();

        try {
            const response = await fetch("/conversations");
            const conversations = await response.json();

            if (conversations.length > 0) {
                await loadConversation(conversations[0].id);
            } else {
                await newChat();
            }

        } catch (error) {
            statusElement.textContent =
                "تعذر تحميل المحادثات.";
        }
    }


    initialize();
</script>
