<script>
    // =====================================================
    // STATE
    // =====================================================

    let currentConversationId = null;
    let isSending = false;

    const providerNames = {
        gemini: "Google Gemini",
        openrouter: "OpenRouter",
        groq: "Groq",
        deepseek: "DeepSeek API"
    };

    const keyInputIds = {
        gemini: "geminiKey",
        openrouter: "openrouterKey",
        groq: "groqKey",
        deepseek: "deepseekKey"
    };

    const keyStatusIds = {
        gemini: "geminiStatus",
        openrouter: "openrouterStatus",
        groq: "groqStatus",
        deepseek: "deepseekStatus"
    };


    // =====================================================
    // HELPERS
    // =====================================================

    async function apiFetch(url, options = {}) {
        const response = await fetch(url, {
            ...options,
            headers: {
                ...(options.body ? {
                    "Content-Type": "application/json"
                } : {}),
                ...(options.headers || {})
            }
        });

        let data;

        try {
            data = await response.json();
        } catch {
            throw new Error("استجابة غير صالحة من الخادم.");
        }

        if (!response.ok) {
            throw new Error(
                data.error || "حدث خطأ أثناء الاتصال بالخادم."
            );
        }

        return data;
    }


    function escapeHtml(value) {
        return String(value)
            .replaceAll("&", "&amp;")
            .replaceAll("<", "&lt;")
            .replaceAll(">", "&gt;")
            .replaceAll('"', "&quot;")
            .replaceAll("'", "&#039;");
    }


    function scrollToBottom() {
        const chatArea = document.getElementById("chatArea");

        chatArea.scrollTop = chatArea.scrollHeight;
    }


    function showWelcome() {
        const messages = document.getElementById("messages");

        messages.innerHTML = `
            <div class="welcome" id="welcome">
                <h1>مرحبًا بك 👋</h1>

                <p>
                    اختر مزود الذكاء الاصطناعي والموديل،
                    ثم ابدأ محادثتك.
                </p>

                <p>
                    تُحفظ المحادثات محليًا في قاعدة البيانات.
                </p>
            </div>
        `;
    }


    function addMessage(role, content, options = {}) {
        const messages = document.getElementById("messages");

        const welcome = document.getElementById("welcome");

        if (welcome) {
            welcome.remove();
        }

        const message = document.createElement("div");

        message.className = `message ${role}`;

        const roleName = role === "user"
            ? "أنت"
            : role === "error"
                ? "خطأ"
                : "المساعد";

        const metadata = options.provider
            ? `${providerNames[options.provider] || options.provider}`
              + (options.model ? ` · ${options.model}` : "")
            : "";

        const header = document.createElement("div");

        header.className = "message-header";

        header.textContent = metadata
            ? `${roleName} — ${metadata}`
            : roleName;

        const body = document.createElement("div");

        body.className = "message-content";

        // Use textContent to prevent HTML injection.
        body.textContent = content;

        message.appendChild(header);
        message.appendChild(body);

        messages.appendChild(message);

        scrollToBottom();

        return message;
    }


    function setModelStatus(message) {
        document.getElementById("modelStatus").textContent = message;
    }


    // =====================================================
    // PROVIDERS AND MODELS
    // =====================================================

    async function providerChanged() {
        const provider = document.getElementById("provider").value;

        const modelSelect = document.getElementById("model");

        const notice = document.getElementById("providerNotice");

        modelSelect.innerHTML = "";

        modelSelect.disabled = true;

        notice.textContent = "";

        setModelStatus("جارٍ تحميل الموديلات...");

        try {
            const data = await apiFetch(
                `/api/models/${encodeURIComponent(provider)}`
            );

            if (data.notice) {
                notice.textContent = data.notice;
            }

            const models = data.models || [];

            if (models.length === 0) {
                modelSelect.innerHTML = "";

                const option = document.createElement("option");

                option.value = "";
                option.textContent = "لا توجد موديلات متاحة";

                modelSelect.appendChild(option);

                setModelStatus(
                    data.error
                        ? data.error
                        : "لا توجد موديلات في القائمة الحالية."
                );

                return;
            }

            for (const model of models) {
                const option = document.createElement("option");

                option.value = model.id;
                option.textContent = model.name || model.id;

                modelSelect.appendChild(option);
            }

            modelSelect.disabled = false;

            setModelStatus(
                `تم تحميل ${models.length} موديل.`
            );

        } catch (error) {
            modelSelect.innerHTML = "";

            const option = document.createElement("option");

            option.value = "";
            option.textContent = "تعذر تحميل الموديلات";

            modelSelect.appendChild(option);

            setModelStatus(error.message);
        }
    }


    // =====================================================
    // CHAT
    // =====================================================

    async function sendMessage() {
        if (isSending) {
            return;
        }

        const input = document.getElementById("messageInput");

        const sendButton = document.getElementById("sendButton");

        const provider = document.getElementById("provider").value;

        const model = document.getElementById("model").value;

        const message = input.value.trim();

        if (!message) {
            return;
        }

        if (!model) {
            addMessage(
                "error",
                "اختر موديلًا متاحًا قبل إرسال الرسالة."
            );
            return;
        }

        isSending = true;

        sendButton.disabled = true;

        sendButton.textContent = "...";

        input.value = "";

        addMessage("user", message);

        const waitingMessage = addMessage(
            "assistant",
            "جارٍ التفكير..."
        );

        try {
            const data = await apiFetch("/chat", {
                method: "POST",

                body: JSON.stringify({
                    message: message,
                    provider: provider,
                    model: model,
                    conversation_id: currentConversationId
                })
            });

            currentConversationId = data.conversation_id;

            waitingMessage.remove();

            addMessage(
                "assistant",
                data.answer,
                {
                    provider: data.provider,
                    model: data.model
                }
            );

            await loadConversations();

        } catch (error) {
            waitingMessage.remove();

            addMessage("error", error.message);

            // If the server created a conversation before
            // the provider failed, keep its ID when supplied.
            // The error response can contain conversation_id.
        } finally {
            isSending = false;

            sendButton.disabled = false;

            sendButton.textContent = "إرسال";

            input.focus();
        }
    }


    // =====================================================
    // CONVERSATIONS
    // =====================================================

    async function loadConversations() {
        try {
            const data = await apiFetch("/conversations");

            const list = document.getElementById("conversationList");

            list.innerHTML = "";

            for (const conversation of data.conversations || []) {
                const item = document.createElement("div");

                item.className = "conversation-item";

                if (
                    Number(conversation.id) ===
                    Number(currentConversationId)
                ) {
                    item.classList.add("active");
                }

                const title = document.createElement("button");

                title.className = "conversation-title";

                title.textContent = conversation.title || "محادثة جديدة";

                title.title = conversation.title || "محادثة جديدة";

                title.onclick = () => {
                    openConversation(conversation.id);
                };

                const deleteButton = document.createElement("button");

                deleteButton.className = "delete-button";

                deleteButton.textContent = "×";

                deleteButton.title = "حذف المحادثة";

                deleteButton.onclick = (event) => {
                    event.stopPropagation();

                    deleteConversation(conversation.id);
                };

                item.appendChild(title);

                item.appendChild(deleteButton);

                list.appendChild(item);
            }

        } catch (error) {
            console.error(error);
        }
    }


    async function newConversation() {
        try {
            const data = await apiFetch("/conversations", {
                method: "POST",
                body: JSON.stringify({})
            });

            currentConversationId = data.id;

            showWelcome();

            await loadConversations();

            closeSidebar();

        } catch (error) {
            addMessage("error", error.message);
        }
    }


    async function openConversation(id) {
        try {
            const data = await apiFetch(
                `/conversations/${encodeURIComponent(id)}`
            );

            currentConversationId = data.conversation.id;

            const messages = document.getElementById("messages");

            messages.innerHTML = "";

            if (!data.messages || data.messages.length === 0) {
                showWelcome();
            } else {
                for (const message of data.messages) {
                    const role = message.role === "user"
                        ? "user"
                        : "assistant";

                    addMessage(
                        role,
                        message.content,
                        {
                            provider: message.provider,
                            model: message.model
                        }
                    );
                }
            }

            await loadConversations();

            closeSidebar();

        } catch (error) {
            addMessage("error", error.message);
        }
    }


    async function deleteConversation(id) {
        const confirmed = confirm(
            "هل تريد حذف هذه المحادثة نهائيًا؟"
        );

        if (!confirmed) {
            return;
        }

        try {
            await apiFetch(
                `/conversations/${encodeURIComponent(id)}`,
                {
                    method: "DELETE"
                }
            );

            if (
                Number(currentConversationId) === Number(id)
            ) {
                currentConversationId = null;

                showWelcome();
            }

            await loadConversations();

        } catch (error) {
            alert(error.message);
        }
    }

    async function openSettings() {
        document.getElementById("settingsModal")
            .classList.add("open");

        document.getElementById("settingsMessage")
            .textContent = "";

        await loadKeyStatus();
    }


    function closeSettings() {
        document.getElementById("settingsModal")
            .classList.remove("open");
    }


    function handleBackdropClick(event) {
        if (event.target.id === "settingsModal") {
            closeSettings();
        }
    }


    async function loadKeyStatus() {
        try {
            const data = await apiFetch("/api/keys");

            for (const provider of Object.keys(keyInputIds)) {
                const input = document.getElementById(
                    keyInputIds[provider]
                );

                const status = document.getElementById(
                    keyStatusIds[provider]
                );

                input.value = "";

                if (data.keys[provider]) {
                    status.textContent = "✓ يوجد مفتاح محفوظ";

                    status.classList.add("saved");
                } else {
                    status.textContent = "لا يوجد مفتاح محفوظ";

                    status.classList.remove("saved");
                }
            }

        } catch (error) {
            showSettingsMessage(error.message, true);
        }
    }


    async function saveKeys() {
        const button = document.getElementById("saveKeysButton");

        const payload = {};

        for (const provider of Object.keys(keyInputIds)) {
            payload[provider] = document.getElementById(
                keyInputIds[provider]
            ).value.trim();
        }

        button.disabled = true;

        button.textContent = "جارٍ الحفظ...";

        try {
            const data = await apiFetch("/api/keys", {
                method: "POST",

                body: JSON.stringify(payload)
            });

            for (const provider of Object.keys(keyInputIds)) {
                document.getElementById(
                    keyInputIds[provider]
                ).value = "";
            }

            showSettingsMessage(
                data.message || "تم حفظ المفاتيح.",
                false
            );

            await loadKeyStatus();

            await providerChanged();

        } catch (error) {
            showSettingsMessage(error.message, true);

        } finally {
            button.disabled = false;

            button.textContent = "حفظ المفاتيح";
        }
    }


    function showSettingsMessage(message, isError) {
        const element = document.getElementById("settingsMessage");

        element.textContent = message;

        element.className = "modal-message " +
            (isError ? "error" : "success");
    }
