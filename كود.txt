<script>
"use strict";


// =========================================================
// State
// =========================================================

let currentConversationId = null;
let isSending = false;
let statusTimer = null;


// =========================================================
// DOM helpers
// =========================================================

const $ = (id) => document.getElementById(id);

const messagesElement = $("messages");
const conversationListElement = $("conversationList");
const providerSelect = $("providerSelect");
const modelSelect = $("modelSelect");
const messageInput = $("messageInput");
const sendButton = $("sendButton");
const welcomeElement = $("welcome");
const pageTitleElement = $("pageTitle");


// =========================================================
// API helpers
// =========================================================

async function apiFetch(url, options = {}) {
    let response;

    try {
        response = await fetch(url, {
            ...options,
            headers: {
                ...(options.body
                    ? {"Content-Type": "application/json"}
                    : {}),
                ...(options.headers || {})
            }
        });
    } catch (error) {
        throw new Error(
            "تعذر الاتصال بخادم Flask. تأكد أن التطبيق يعمل."
        );
    }

    const contentType =
        response.headers.get("content-type") || "";

    let data;

    if (contentType.includes("application/json")) {
        try {
            data = await response.json();
        } catch (error) {
            throw new Error(
                `تعذر قراءة استجابة الخادم (HTTP ${response.status}).`
            );
        }
    } else {
        const responseText = await response.text();

        console.error(
            "Unexpected server response:",
            response.status,
            responseText.slice(0, 2000)
        );

        throw new Error(
            `أعاد الخادم استجابة غير متوقعة (HTTP ${response.status}).`
            + " افتح نافذة CMD لمعرفة سبب الخطأ."
        );
    }

    if (!response.ok) {
        const error = new Error(
            data.error || `حدث خطأ في الخادم (${response.status}).`
        );

        error.status = response.status;
        error.data = data;

        throw error;
    }

    return data;
}


function showStatus(message, isError = false) {
    const element = $("status");

    element.textContent = message;
    element.className = "status show";

    if (isError) {
        element.classList.add("error");
    }

    clearTimeout(statusTimer);

    statusTimer = setTimeout(() => {
        element.className = "status";
    }, 6000);
}


// =========================================================
// Safe message rendering
// =========================================================

function createMessageElement(message) {
    const wrapper = document.createElement("div");

    wrapper.className =
        "message " +
        (message.role === "user" ? "user" : "assistant");

    const role = document.createElement("div");

    role.className = "message-role";

    role.textContent =
        message.role === "user"
            ? "أنت"
            : "المساعد";

    const content = document.createElement("div");

    content.className = "message-content";

    // Use textContent to avoid interpreting model output as HTML.
    content.textContent = message.content || "";

    wrapper.appendChild(role);
    wrapper.appendChild(content);

    if (message.provider || message.model) {
        const meta = document.createElement("div");

        meta.className = "message-meta";

        const parts = [];

        if (message.provider) {
            parts.push(message.provider);
        }

        if (message.model) {
            parts.push(message.model);
        }

        meta.textContent = parts.join(" · ");

        wrapper.appendChild(meta);
    }

    return wrapper;
}


function renderMessage(message) {
    welcomeElement.style.display = "none";

    const element = createMessageElement(message);

    messagesElement.appendChild(element);

    messagesElement.scrollTop =
        messagesElement.scrollHeight;

    return element;
}


function showWelcome() {
    messagesElement.replaceChildren();

    messagesElement.appendChild(welcomeElement);

    welcomeElement.style.display = "block";

    pageTitleElement.textContent =
        "مساعد الذكاء الاصطناعي";
}


function renderMessages(messageList) {
    messagesElement.replaceChildren();

    for (const message of messageList) {
        renderMessage(message);
    }

    if (messageList.length === 0) {
        messagesElement.appendChild(welcomeElement);
        welcomeElement.style.display = "block";
    }

    messagesElement.scrollTop =
        messagesElement.scrollHeight;
}


// =========================================================
// Conversations
// =========================================================

async function loadConversations() {
    const data = await apiFetch("/conversations");

    const conversations = data.conversations || [];

    conversationListElement.replaceChildren();

    for (const conversation of conversations) {
        const item = document.createElement("div");

        item.className = "conversation-item";

        if (
            Number(conversation.id) ===
            Number(currentConversationId)
        ) {
            item.classList.add("active");
        }

        const openButton = document.createElement("button");

        openButton.type = "button";
        openButton.className = "conversation-open";

        openButton.textContent =
            conversation.title || "محادثة جديدة";

        openButton.title = openButton.textContent;

        openButton.addEventListener("click", () => {
            openConversation(conversation.id);
        });

        const deleteButton = document.createElement("button");

        deleteButton.type = "button";
        deleteButton.className = "delete-button";
        deleteButton.textContent = "×";
        deleteButton.title = "حذف المحادثة";

        deleteButton.addEventListener("click", (event) => {
            event.stopPropagation();

            deleteConversation(conversation.id);
        });

        item.appendChild(openButton);
        item.appendChild(deleteButton);

        conversationListElement.appendChild(item);
    }
}


async function newConversation() {
    if (isSending) {
        showStatus(
            "انتظر حتى تنتهي الرسالة الحالية.",
            true
        );

        return;
    }

    try {
        const data = await apiFetch(
            "/conversations",
            {
                method: "POST",
                body: JSON.stringify({})
            }
        );

        const conversation = data.conversation;

        const id = Number(
            conversation ? conversation.id : data.id
        );

        if (!Number.isInteger(id) || id <= 0) {
            throw new Error(
                "أنشأ الخادم استجابة غير مكتملة للمحادثة."
            );
        }

        currentConversationId = id;

        showWelcome();

        await loadConversations();

        messageInput.focus();

        showStatus("تم إنشاء محادثة جديدة.");

    } catch (error) {
        console.error("New conversation error:", error);

        showStatus(error.message, true);
    }
}


async function openConversation(id) {
    if (isSending) {
        showStatus(
            "انتظر حتى تنتهي الرسالة الحالية.",
            true
        );

        return;
    }

    try {
        const data = await apiFetch(
            `/conversations/${encodeURIComponent(id)}`
        );

        if (!data.conversation) {
            throw new Error(
                "لم يُرجع الخادم بيانات المحادثة."
            );
        }

        currentConversationId =
            Number(data.conversation.id);

        pageTitleElement.textContent =
            data.conversation.title || "محادثة جديدة";

        renderMessages(data.messages || []);

        await loadConversations();

    } catch (error) {
        console.error("Open conversation error:", error);

        showStatus(error.message, true);

        try {
            await loadConversations();
        } catch (refreshError) {
            console.error(refreshError);
        }
    }
}


async function deleteConversation(id) {
    if (isSending) {
        showStatus(
            "انتظر حتى تنتهي الرسالة الحالية قبل الحذف.",
            true
        );

        return;
    }

    const confirmed = confirm(
        "هل تريد حذف هذه المحادثة وجميع رسائلها؟"
    );

    if (!confirmed) {
        return;
    }

    const deletedId = Number(id);

    try {
        await apiFetch(
            `/conversations/${encodeURIComponent(deletedId)}`,
            {
                method: "DELETE"
            }
        );

        if (
            Number(currentConversationId) === deletedId
        ) {
            currentConversationId = null;

            showWelcome();
        }

        await loadConversations();

        showStatus("تم حذف المحادثة.");

    } catch (error) {
        console.error("Delete conversation error:", error);

        showStatus(error.message, true);
    }
}


// =========================================================
// Provider and model selection
// =========================================================

async function loadModels(provider) {
    modelSelect.replaceChildren();

    const placeholder = document.createElement("option");

    placeholder.value = "";
    placeholder.textContent = "جاري تحميل النماذج...";

    modelSelect.appendChild(placeholder);

    modelSelect.disabled = true;

    try {
        const data = await apiFetch(
            `/api/models/${encodeURIComponent(provider)}`
        );

        modelSelect.replaceChildren();

        const models = data.models || [];

        if (models.length === 0) {
            const option = document.createElement("option");

            option.value = "";
            option.textContent = "لا توجد نماذج متاحة";

            modelSelect.appendChild(option);

            showStatus(
                "لم يتم العثور على نماذج لهذا المزود.",
                true
            );

            return;
        }

        const firstOption = document.createElement("option");

        firstOption.value = "";
        firstOption.textContent = "اختر النموذج";

        modelSelect.appendChild(firstOption);

        for (const model of models) {
            const option = document.createElement("option");

            option.value = model.id;
            option.textContent = model.name || model.id;

            modelSelect.appendChild(option);
        }

        // Select the first available model automatically.
        modelSelect.value = models[0].id;

    } catch (error) {
        console.error("Load models error:", error);

        modelSelect.replaceChildren();

        const option = document.createElement("option");

        option.value = "";
        option.textContent = "تعذر تحميل النماذج";

        modelSelect.appendChild(option);

        showStatus(error.message, true);

    } finally {
        modelSelect.disabled = false;
    }
}


providerSelect.addEventListener("change", async () => {
    await loadModels(providerSelect.value);
});


// =========================================================
// Chat
// =========================================================

async function sendMessage() {
    if (isSending) {
        return;
    }

    const message = messageInput.value.trim();

    if (!message) {
        return;
    }

    const provider = providerSelect.value;
    const model = modelSelect.value;

    if (!model) {
        showStatus(
            "اختر نموذجًا أولًا. إذا لم تظهر النماذج، تحقق من مفتاح API.",
            true
        );

        return;
    }

    isSending = true;

    sendButton.disabled = true;
    sendButton.textContent = "جارٍ الإرسال...";

    messageInput.disabled = true;

    const submittedMessage = message;

    messageInput.value = "";
    messageInput.style.height = "auto";

    renderMessage({
        role: "user",
        content: submittedMessage,
        provider: provider,
        model: model
    });

    const typing = document.createElement("div");

    typing.className = "message assistant typing";
    typing.textContent = "يفكر المساعد...";

    messagesElement.appendChild(typing);

    messagesElement.scrollTop =
        messagesElement.scrollHeight;

    try {
        const data = await apiFetch(
            "/chat",
            {
                method: "POST",
                body: JSON.stringify({
                    conversation_id: currentConversationId,
                    message: submittedMessage,
                    provider: provider,
                    model: model
                })
            }
        );

        typing.remove();

        if (data.conversation_id) {
            currentConversationId =
                Number(data.conversation_id);
        }

        renderMessage({
            role: "assistant",
            content: data.reply || "",
            provider: data.provider || provider,
            model: data.model || model
        });

        pageTitleElement.textContent =
            data.conversation?.title || "محادثة";

        await loadConversations();

    } catch (error) {
        console.error("Chat error:", error);

        typing.remove();

        /*
         * Critical recovery:
         * If the server created a conversation but the provider
         * failed, keep the conversation ID returned in the error.
         */

        if (
            error.data &&
            error.data.conversation_id
        ) {
            currentConversationId =
                Number(error.data.conversation_id);

            try {
                await loadConversations();
            } catch (refreshError) {
                console.error(refreshError);
            }
        }

        renderMessage({
            role: "assistant",
            content: "تعذر الحصول على الإجابة: " + error.message,
            provider: provider,
            model: model
        });

        showStatus(error.message, true);

    } finally {
        isSending = false;

        sendButton.disabled = false;
        sendButton.textContent = "إرسال";

        messageInput.disabled = false;

        messageInput.focus();
    }
}


// =========================================================
// API key settings
// =========================================================

function openSettings() {
    $("settingsModal").classList.add("open");

    $("keyProvider").value = providerSelect.value;

    $("apiKeyInput").value = "";

    updateKeyStatus();
}


function closeSettings() {
    $("settingsModal").classList.remove("open");
}


async function updateKeyStatus() {
    const provider = $("keyProvider").value;

    $("keyStatus").textContent =
        "جارٍ التحقق من إعدادات المفتاح...";

    try {
        const data = await apiFetch("/api/keys");

        const configured =
            Boolean(data.keys && data.keys[provider]);

        $("keyStatus").textContent = configured
            ? "يوجد مفتاح محفوظ لهذا المزود. أدخل مفتاحًا جديدًا لاستبداله."
            : "لم يتم حفظ مفتاح لهذا المزود بعد.";

    } catch (error) {
        $("keyStatus").textContent = error.message;
    }
}


async function saveApiKey() {
    const provider = $("keyProvider").value;
    const apiKey = $("apiKeyInput").value.trim();

    if (!apiKey) {
        showStatus(
            "أدخل مفتاح API أولًا.",
            true
        );

        return;
    }

    const button = $("saveKeyButton");

    button.disabled = true;
    button.textContent = "جارٍ الحفظ...";

    try {
        await apiFetch(
            "/api/keys",
            {
                method: "POST",
                body: JSON.stringify({
                    provider: provider,
                    api_key: apiKey
                })
            }
        );

        $("apiKeyInput").value = "";

        showStatus("تم حفظ مفتاح API بنجاح.");

        await updateKeyStatus();

        if (provider === providerSelect.value) {
            await loadModels(provider);
        }

    } catch (error) {
        console.error("Save API key error:", error);

        showStatus(error.message, true);

    } finally {
        button.disabled = false;
        button.textContent = "حفظ المفتاح";
    }
}

            true

    await loadModels(providerSelect.value);

}
