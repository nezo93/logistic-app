# ==================================================
# API KEYS
# ==================================================

def load_keys():
    keys = {provider: "" for provider in PROVIDERS}

    if os.path.isfile(KEYS_FILE):
        try:
            with open(KEYS_FILE, "r", encoding="utf-8") as file:
                saved = json.load(file)

            for provider in PROVIDERS:
                keys[provider] = str(
                    saved.get(provider, "")
                ).strip()

        except (OSError, json.JSONDecodeError):
            pass

    # متغيرات البيئة بديلة إذا لم يوجد مفتاح محفوظ.
    for provider, settings in PROVIDERS.items():
        if not keys[provider]:
            keys[provider] = os.environ.get(
                settings["env"], ""
            ).strip()

    return keys


def get_key(provider):
    return load_keys().get(provider, "")


def save_keys(keys):
    temporary_file = KEYS_FILE + ".tmp"

    with open(temporary_file, "w", encoding="utf-8") as file:
        json.dump(keys, file, ensure_ascii=False, indent=2)

    os.replace(temporary_file, KEYS_FILE)

    try:
        os.chmod(KEYS_FILE, 0o600)
    except OSError:
        pass


@app.route("/api/keys", methods=["GET"])
def api_keys_status():
    keys = load_keys()

    return jsonify({
        provider: {
            "name": settings["name"],
            "configured": bool(keys[provider])
        }
        for provider, settings in PROVIDERS.items()
    })


@app.route("/api/keys", methods=["POST"])
def api_keys_save():
    data = request.get_json(silent=True) or {}
    keys = load_keys()

    for provider in PROVIDERS:
        if provider in data:
            value = str(data[provider]).strip()

            # الحقل الفارغ لا يمسح المفتاح الحالي.
            if value:
                keys[provider] = value

    try:
        save_keys(keys)
    except OSError:
        return jsonify({
            "error": "تعذر حفظ المفاتيح على الجهاز."
        }), 500

    return jsonify({
        "success": True,
        "message": "تم حفظ المفاتيح بنجاح."
    })

def get_db():
    conn = sqlite3.connect(DB_FILE, timeout=30)
    conn.row_factory = sqlite3.Row
    return conn


def init_db():
    with get_db() as conn:
        conn.execute("""
            CREATE TABLE IF NOT EXISTS conversations (
                id TEXT PRIMARY KEY,
                title TEXT NOT NULL,
                created_at TEXT NOT NULL,
                updated_at TEXT NOT NULL
            )
        """)

        conn.execute("""
            CREATE TABLE IF NOT EXISTS messages (
                id INTEGER PRIMARY KEY AUTOINCREMENT,
                conversation_id TEXT NOT NULL,
                role TEXT NOT NULL,
                content TEXT NOT NULL,
                model TEXT,
                created_at TEXT NOT NULL
            )
        """)


def now():
    return datetime.now().isoformat(timespec="seconds")


def create_conversation(title="محادثة جديدة"):
    conversation_id = str(uuid.uuid4())
    timestamp = now()

    with get_db() as conn:
        conn.execute("""
            INSERT INTO conversations
                (id, title, created_at, updated_at)
            VALUES (?, ?, ?, ?)
        """, (conversation_id, title, timestamp, timestamp))

    return conversation_id


def conversation_exists(conversation_id):
    with get_db() as conn:
        row = conn.execute(
            "SELECT id FROM conversations WHERE id = ?",
            (conversation_id,)
        ).fetchone()

    return row is not None


def save_message(conversation_id, role, content, model=None):
    timestamp = now()

    with get_db() as conn:
        conn.execute("""
            INSERT INTO messages
                (conversation_id, role, content, model, created_at)
            VALUES (?, ?, ?, ?, ?)
        """, (
            conversation_id,
            role,
            content,
            model,
            timestamp
        ))

        conn.execute("""
            UPDATE conversations
            SET updated_at = ?
            WHERE id = ?
        """, (timestamp, conversation_id))


def get_messages(conversation_id):
    with get_db() as conn:
        rows = conn.execute("""
            SELECT role, content, model, created_at
            FROM messages
            WHERE conversation_id = ?
            ORDER BY id
        """, (conversation_id,)).fetchall()

    return [dict(row) for row in rows]

def get_models(provider):
    key = get_key(provider)

    if not key:
        return []

    try:
        if provider == "gemini":
            response = requests.get(
                "https://generativelanguage.googleapis.com/v1beta/models",
                params={
                    "key": key,
                    "pageSize": 1000
                },
                timeout=30
            )

            if not response.ok:
                print("Gemini models error:", response.text[:500])
                return []

            result = []

            for model in response.json().get("models", []):
                methods = model.get(
                    "supportedGenerationMethods", []
                )

                if "generateContent" not in methods:
                    continue

                model_id = model.get("name", "").replace(
                    "models/", "", 1
                )

                if model_id:
                    result.append({
                        "id": model_id,
                        "name": model.get(
                            "displayName", model_id
                        )
                    })

            return result

        if provider == "anthropic":
            response = requests.get(
                "https://api.anthropic.com/v1/models",
                headers={
                    "x-api-key": key,
                    "anthropic-version": "2023-06-01"
                },
                params={"limit": 100},
                timeout=30
            )

            if not response.ok:
                print("Anthropic models error:", response.text[:500])
                return []

            return [
                {
                    "id": model["id"],
                    "name": model.get(
                        "display_name", model["id"]
                    )
                }
                for model in response.json().get("data", [])
                if model.get("id")
            ]

        if provider == "openai":
            response = requests.get(
                "https://api.openai.com/v1/models",
                headers={
                    "Authorization": f"Bearer {key}"
                },
                timeout=30
            )

            if not response.ok:
                print("OpenAI models error:", response.text[:500])
                return []

            result = []

            for model in response.json().get("data", []):
                model_id = model.get("id", "")

                # عرض موديلات GPT وموديلات الاستدلال الشائعة.
                # ليس كل موديل في هذه القائمة مضمون التوافق
                # مع واجهة chat/completions.
                if model_id.startswith(
                    ("gpt-", "o1", "o3", "o4")
                ):
                    result.append({
                        "id": model_id,
                        "name": model_id
                    })

            return result

    except requests.RequestException as error:
        print(f"Model discovery error ({provider}): {error}")

    return []


@app.route("/api/models/<provider>", methods=["GET"])
def models_route(provider):
    if provider not in PROVIDERS:
        return jsonify({
            "error": "المزوّد غير معروف."
        }), 400

    key = get_key(provider)

    if not key:
        return jsonify({
            "provider": provider,
            "configured": False,
            "models": [],
            "error": "أضف مفتاح API أولاً."
        })

    models = get_models(provider)

    return jsonify({
        "provider": provider,
        "configured": True,
        "models": models
    })

def call_ai(provider, model, history, message):
    key = get_key(provider)

    if not key:
        raise ValueError(
            f"لم تتم إضافة مفتاح {PROVIDERS[provider]['name']}."
        )

    # لا نكرر الرسالة الحالية في السجل؛ فهي تُرسل منفصلة.
    chat_history = [
        {
            "role": item["role"],
            "content": item["content"]
        }
        for item in history
        if item["role"] in ("user", "assistant")
    ]

    if provider == "gemini":
        client = genai.Client(api_key=key)

        contents = []

        for item in chat_history:
            contents.append({
                "role": (
                    "user"
                    if item["role"] == "user"
                    else "model"
                ),
                "parts": [{"text": item["content"]}]
            })

        contents.append({
            "role": "user",
            "parts": [{"text": message}]
        })

        response = client.models.generate_content(
            model=model,
            contents=contents
        )

        return response.text or "لم يصل رد نصي من النموذج."

    if provider == "anthropic":
        # Claude لا يقبل system كرسالة عادية داخل messages.
        messages = [
            {
                "role": item["role"],
                "content": item["content"]
            }
            for item in chat_history
        ]

        messages.append({
            "role": "user",
            "content": message
        })

        response = requests.post(
            "https://api.anthropic.com/v1/messages",
            headers={
                "x-api-key": key,
                "anthropic-version": "2023-06-01",
                "Content-Type": "application/json"
            },
            json={
                "model": model,
                "max_tokens": 4096,
                "messages": messages
            },
            timeout=120
        )

        data = response.json()

        if not response.ok:
            raise ValueError(
                data.get("error", {}).get(
                    "message", str(data)
                )
            )

        return "".join(
            block.get("text", "")
            for block in data.get("content", [])
            if block.get("type") == "text"
        )

    if provider == "openai":
        messages = list(chat_history)

        messages.append({
            "role": "user",
            "content": message
        })

        response = requests.post(
            "https://api.openai.com/v1/chat/completions",
            headers={
                "Authorization": f"Bearer {key}",
                "Content-Type": "application/json"
            },
            json={
                "model": model,
                "messages": messages
            },
            timeout=120
        )

        data = response.json()

        if not response.ok:
            raise ValueError(
                data.get("error", {}).get(
                    "message", str(data)
                )
            )

        return data["choices"][0]["message"]["content"]

    raise ValueError("المزوّد غير مدعوم.")

@app.route("/conversations/<conversation_id>", methods=["GET"])

        return jsonify({
            "error": "المحادثة غير موجودة"

    return jsonify({

@app.route("/conversations/<conversation_id>", methods=["DELETE"])
def delete_conversation(conversation_id):
    with get_db() as conn:
        conn.execute(
            "DELETE FROM messages WHERE conversation_id = ?",
            (conversation_id,)
        )

        conn.execute(
            "DELETE FROM conversations WHERE id = ?",
            (conversation_id,)
        )

    return jsonify({"success": True})


@app.route("/chat", methods=["POST"])
def chat():
    data = request.get_json(silent=True) or {}

    message = str(data.get("message", "")).strip()
    model = str(data.get("model", "")).strip()
    provider = str(data.get("provider", "gemini")).strip()
    conversation_id = data.get("conversation_id")

    if not message:
        return jsonify({
            "error": "اكتب رسالة أولاً."
        }), 400

    if provider not in PROVIDERS:
        return jsonify({
            "error": "المزوّد غير معروف."
        }), 400

    if not model:
        return jsonify({
            "error": "اختر نموذجاً أولاً."
        }), 400

    if not get_key(provider):
        return jsonify({
            "error": (
                f"أضف مفتاح {PROVIDERS[provider]['name']} أولاً."
            )
        }), 400

    if not conversation_id or not conversation_exists(
        conversation_id
    ):
        conversation_id = create_conversation(message[:60])

    # نحفظ رسالة المستخدم قبل إرسال الطلب.
    save_message(
        conversation_id,
        "user",
        message,
        f"{provider}:{model}"
    )

    # سجل الرسائل يحتوي الرسالة الحالية أيضاً.
    history = get_messages(conversation_id)[:-1]

    try:
        reply = call_ai(
            provider,
            model,
            history,
            message
        )

        save_message(
            conversation_id,
            "assistant",
            reply,
            f"{provider}:{model}"
        )

        return jsonify({
            "reply": reply,
            "provider": provider,
            "model": model,
            "conversation_id": conversation_id
        })

    except Exception as error:
        app.logger.exception("AI request failed")

        return jsonify({
            "error": str(error),
            "conversation_id": conversation_id
        }), 500
