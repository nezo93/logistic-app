<script>
    "use strict";

    let currentConversationId = null;
    let isGenerating = false;
    let currentController = null;
    let conversations = [];
    let loadedKeys = {
        gemini: false,
        anthropic: false,
        openai: false
    };

    const $ = id => document.getElementById(id);

    const providerNames = {
        gemini: "Google Gemini",
        anthropic: "Claude",
        openai: "OpenAI"
    };

    /*
     * General API helper
     */

    async function apiRequest(url, options = {}) {
        const response = await fetch(url, {
            ...options,
            headers: {
                ...(options.body ? {"Content-Type": "application/json"} : {}),
                ...(options.headers || {})
            }
        });

        const raw = await response.text();

        let data = {};

        try {
            data = raw ? JSON.parse(raw) : {};
        } catch {
            data = {error: raw || "استجابة غير صالحة من الخادم"};
        }

        if (!response.ok) {
            throw new Error(
                data.error ||
                data.message ||
                data.detail ||
                `HTTP ${response.status}`
            );
        }

        return data;
    }


    /*
     * Settings
     */

    function openSettings() {
        $("settingsOverlay").classList.add("open");
        $("settingsStatus").textContent = "";
        $("settingsStatus").className = "settings-status";

        loadKeyStatus();
    }

    function closeSettings() {
        $("settingsOverlay").classList.remove("open");
    }

    function overlayClick(event) {
        if (event.target === $("settingsOverlay")) {
            closeSettings();
        }
    }

    async function loadKeyStatus() {
        try {
            const data = await apiRequest("/api/keys");

            const keyInfo = data.keys || data.providers || data;

            loadedKeys.gemini = Boolean(
                keyInfo.gemini ||
                keyInfo.GEMINI_API_KEY
            );

            loadedKeys.anthropic = Boolean(
                keyInfo.anthropic ||
                keyInfo.claude ||
                keyInfo.ANTHROPIC_API_KEY
            );

            loadedKeys.openai = Boolean(
                keyInfo.openai ||
                keyInfo.OPENAI_API_KEY
            );

            updateKeyStatuses();

        } catch (error) {
            $("settingsStatus").textContent =
                "تعذر قراءة حالة المفاتيح: " + error.message;

            $("settingsStatus").className =
                "settings-status status-error";
        }
    }

    function updateKeyStatuses() {
        const statuses = [
            ["geminiStatus", "gemini"],
            ["claudeStatus", "anthropic"],
            ["openaiStatus", "openai"]
        ];

        statuses.forEach(([elementId, provider]) => {
            const element = $(elementId);

            element.textContent = loadedKeys[provider]
                ? "✓ يوجد مفتاح محفوظ"
                : "لم يتم العثور على مفتاح محفوظ";

            element.style.color = loadedKeys[provider]
                ? "#22c55e"
                : "#9299a8";
        });
    }

    async function saveApiKeys() {
        const button = $("saveKeysButton");
        const status = $("settingsStatus");

        const gemini = $("geminiKey").value.trim();
        const anthropic = $("claudeKey").value.trim();
        const openai = $("openaiKey").value.trim();

        if (!gemini && !anthropic && !openai) {
            status.textContent =
                "أدخل مفتاحًا واحدًا على الأقل قبل الحفظ.";

            status.className = "settings-status status-error";
            return;
        }

        button.disabled = true;
        button.textContent = "جارٍ الحفظ...";

        status.textContent = "";
        status.className = "settings-status";

        try {
            /*
             * أسماء الحقول متوافقة مع app.py:
             * gemini, anthropic, openai
             */
            const payload = {};

            if (gemini) payload.gemini = gemini;
            if (anthropic) payload.anthropic = anthropic;
            if (openai) payload.openai = openai;

            await apiRequest("/api/keys", {
                method: "POST",
                body: JSON.stringify(payload)
            });

            $("geminiKey").value = "";
            $("claudeKey").value = "";
            $("openaiKey").value = "";

            await loadKeyStatus();

            status.textContent =
                "تم إرسال المفاتيح للحفظ. يمكنك الآن اختيار المزوّد والموديل.";

            status.className = "settings-status status-success";

            await loadModels();

        } catch (error) {
            status.textContent =
                "فشل حفظ المفاتيح: " + error.message;

            status.className = "settings-status status-error";

        } finally {
            button.disabled = false;
            button.textContent = "حفظ المفاتيح";
        }
    }


    /*
     * Provider and models
     */

    function providerChanged() {
        loadModels();
    }

    async function loadModels() {
        const provider = $("provider").value;
        const modelSelect = $("model");

        modelSelect.innerHTML = "";

        const loadingOption = document.createElement("option");
        loadingOption.value = "";
        loadingOption.textContent = "جارٍ تحميل الموديلات...";
        modelSelect.appendChild(loadingOption);

        try {
            const data = await apiRequest(
                "/api/models/" + encodeURIComponent(provider)
            );

            let models = [];

            if (Array.isArray(data)) {
                models = data;
            } else if (Array.isArray(data.models)) {
                models = data.models;
            } else if (Array.isArray(data.data)) {
                models = data.data;
            }

            modelSelect.innerHTML = "";

            if (models.length === 0) {
                const option = document.createElement("option");
                option.value = "";
                option.textContent = "لا توجد موديلات متاحة";
                modelSelect.appendChild(option);

                showNotice(
                    "لم يعثر الخادم على موديلات لهذا المزوّد. تأكد من حفظ المفتاح وصحة إعدادات app.py.",
                    "error"
                );

                return;
            }

            models.forEach(item => {
                const modelId = typeof item === "string"
                    ? item
                    : (item.id || item.name || item.model);

                if (!modelId) return;

                const option = document.createElement("option");

                option.value = modelId;

                option.textContent = typeof item === "object"
                    ? (item.display_name || item.displayName || modelId)
                    : modelId;

                modelSelect.appendChild(option);
            });

            if (modelSelect.options.length > 0) {
                modelSelect.selectedIndex = 0;
            }

        } catch (error) {
            modelSelect.innerHTML = "";

            const option = document.createElement("option");
            option.value = "";
            option.textContent = "تعذر تحميل الموديلات";

            modelSelect.appendChild(option);

            showNotice(
                "خطأ تحميل الموديلات: " + error.message,
                "error"
            );
        }
    }


    /*
     * Conversation management
     */

    async function refreshConversations() {
        try {
            const data = await apiRequest("/conversations");

            conversations = Array.isArray(data)
                ? data
                : (data.conversations || data.items || []);

            renderConversations();

        } catch (error) {
            console.error("Conversation loading error:", error);
        }
    }

    function renderConversations() {
        const list = $("conversationList");
        list.innerHTML = "";

        conversations.forEach(conversation => {
            const id = conversation.id;
            const title = conversation.title || "محادثة جديدة";

            const item = document.createElement("div");

            item.className = "conversation-item" +
                (String(id) === String(currentConversationId)
                    ? " active"
                    : "");

            const name = document.createElement("span");
            name.className = "conversation-name";
            name.textContent = title;

            name.onclick = () => loadConversation(id);

            const deleteButton = document.createElement("button");
            deleteButton.className = "delete-conversation";
            deleteButton.textContent = "×";
            deleteButton.title = "حذف المحادثة";

            deleteButton.onclick = event => {
                event.stopPropagation();
                deleteConversation(id);
            };

            item.appendChild(name);
            item.appendChild(deleteButton);

            item.onclick = () => loadConversation(id);

            list.appendChild(item);
        });
    }

    async function newChat() {
        currentConversationId = null;

        $("messages").innerHTML = "";

        const welcome = document.createElement("div");
        welcome.className = "welcome";
        welcome.id = "welcome";

        welcome.innerHTML = `
            <div class="welcome-icon">✦</div>
            <h1>كيف يمكنني مساعدتك؟</h1>
            <p>
                اطرح سؤالك وابدأ محادثة جديدة مع مساعدك الذكي.
            </p>
        `;

        $("messages").appendChild(welcome);

        $("messageInput").value = "";
        $("messageInput").focus();

        renderConversations();

        if (window.innerWidth <= 760) {
            $("sidebar").classList.remove("open");
        }
    }

    async function loadConversation(id) {
        try {
            const data = await apiRequest(
                "/conversations/" + encodeURIComponent(id)
            );

            currentConversationId = id;

            $("messages").innerHTML = "";

            const messages = Array.isArray(data)
                ? data
                : (data.messages || []);

            messages.forEach(message => {
                appendMessage(
                    message.role,
                    message.content,
                    message.model
                );
            });

            renderConversations();
            scrollToBottom();

            if (window.innerWidth <= 760) {
                $("sidebar").classList.remove("open");
            }

        } catch (error) {
            showNotice(
                "تعذر تحميل المحادثة: " + error.message,
                "error"
            );
        }
    }

    async function deleteConversation(id) {
        if (!confirm("هل تريد حذف هذه المحادثة؟")) return;

        try {
            await apiRequest(
                "/conversations/" + encodeURIComponent(id),
                {method: "DELETE"}
            );

            if (String(id) === String(currentConversationId)) {
                await newChat();
            }

            await refreshConversations();

        } catch (error) {
            showNotice(
                "تعذر حذف المحادثة: " + error.message,
                "error"
            );
        }
    }


    /*
     * Chat
     */

    async function sendMessage() {
        if (isGenerating) return;

        const input = $("messageInput");
        const message = input.value.trim();

        const provider = $("provider").value;
        const model = $("model").value;

        if (!message) return;

        if (!model) {
            showNotice(
                "لا يوجد موديل محدد. تحقق من المفتاح ثم أعد تحميل الموديلات.",
                "error"
            );

            return;
        }

        isGenerating = true;

        $("sendButton").style.display = "none";
        $("stopButton").style.display = "inline-block";

        input.value = "";
        autoResize(input);

        const welcome = $("welcome");

        if (welcome) {
            welcome.remove();
        }

        appendMessage("user", message);

        const typing = document.createElement("div");
        typing.className = "typing";
        typing.textContent = "جارٍ التفكير...";
        typing.id = "typingIndicator";

        $("messages").appendChild(typing);
        scrollToBottom();

        currentController = new AbortController();

        try {
            const payload = {
                message: message,
                provider: provider,
                model: model,
                conversation_id: currentConversationId
            };

            const response = await fetch("/chat", {
                method: "POST",
                headers: {
                    "Content-Type": "application/json"
                },
                body: JSON.stringify(payload),
                signal: currentController.signal
            });

            const raw = await response.text();

            let data;

            try {
                data = raw ? JSON.parse(raw) : {};
            } catch {
                throw new Error(
                    raw || "استجابة غير صالحة من الخادم"
                );
            }

            if (!response.ok) {
                throw new Error(
                    data.error ||
                    data.message ||
                    data.detail ||
                    "HTTP " + response.status
                );
            }

            if (data.error) {
                throw new Error(data.error);
            }

            $("typingIndicator")?.remove();

            const answer =
                data.response ??
                data.answer ??
                data.reply ??
                data.content ??
                data.message ??
                "";

            if (!answer) {
                throw new Error(
                    "وصل رد فارغ من الخادم. تحقق من سجلات Flask."
                );
            }

            if (data.conversation_id !== undefined &&
                data.conversation_id !== null) {
                currentConversationId = data.conversation_id;
            }

            appendMessage(
                "assistant",
                String(answer),
                data.model || model
            );

            await refreshConversations();

        } catch (error) {
            $("typingIndicator")?.remove();

            if (error.name === "AbortError") {
                showNotice("تم إيقاف الطلب من المتصفح.", "error");
            } else {
                showNotice(
                    "حدث خطأ: " + error.message,
                    "error"
                );
            }

        } finally {
            isGenerating = false;
            currentController = null;

            $("sendButton").style.display = "inline-block";
            $("stopButton").style.display = "none";

            input.focus();
            scrollToBottom();
        }
    }

    function stopGeneration() {
        if (currentController) {
            currentController.abort();
        }
    }

    function handleKey(event) {
        if (event.key === "Enter" && !event.shiftKey) {
            event.preventDefault();
            sendMessage();
        }
    }


    /*
     * Display messages safely
     */

    function appendMessage(role, content, model = "") {
        const welcome = $("welcome");

        if (welcome) {
            welcome.remove();
        }

        const message = document.createElement("div");

        message.className =
            "message " + (role === "user" ? "user" : "assistant");

        const roleLabel = document.createElement("div");
        roleLabel.className = "message-role";

        if (role === "user") {
            roleLabel.textContent = "أنت";
        } else {
            roleLabel.textContent = model
                ? "المساعد · " + String(model).split("/").pop()
                : "المساعد";
        }

        const contentDiv = document.createElement("div");
        contentDiv.className = "message-content";

        if (role === "assistant") {
            renderAssistantContent(contentDiv, String(content));
        } else {
            contentDiv.textContent = String(content);
        }

        message.appendChild(roleLabel);
        message.appendChild(contentDiv);

        $("messages").appendChild(message);

        scrollToBottom();

        return message;
    }

    function renderAssistantContent(container, text) {
        /*
         * Render fenced code blocks without inserting raw HTML.
         */
        const parts = text.split(/```([^\n`]*)\n([\s\S]*?)```/g);

        if (parts.length === 1) {
            renderPlainText(container, text);
            return;
        }

        for (let i = 0; i < parts.length; i++) {
            if (i % 3 === 0) {
                if (parts[i]) {
                    renderPlainText(container, parts[i]);
                }
            } else if (i % 3 === 2) {
                const code = parts[i];

                const pre = document.createElement("pre");
                const codeElement = document.createElement("code");

                codeElement.textContent = code;

                const copyButton = document.createElement("button");
                copyButton.className = "code-copy";
                copyButton.textContent = "نسخ";

                copyButton.onclick = async () => {
                    try {
                        await navigator.clipboard.writeText(code);
                        copyButton.textContent = "تم النسخ";
                    } catch {
                        copyButton.textContent = "تعذر النسخ";
                    }

                    setTimeout(() => {
                        copyButton.textContent = "نسخ";
                    }, 1500);
                };

                pre.appendChild(codeElement);
                pre.appendChild(copyButton);

                container.appendChild(pre);
            }
        }
    }

    function renderPlainText(container, text) {
        const lines = text.split("\n");

        lines.forEach((line, index) => {
            const paragraph = document.createElement("p");

            paragraph.textContent = line || "\u00a0";

            container.appendChild(paragraph);
        });
    }

    function showNotice(message, type = "error") {
        const notice = document.createElement("div");

        notice.className = type === "error"
            ? "error-message"
            : "typing";

        notice.textContent = message;

        $("messages").appendChild(notice);

        scrollToBottom();
    }

    function scrollToBottom() {
        const container = $("chatContainer");

        container.scrollTop = container.scrollHeight;
    }

    function autoResize(textarea) {
        textarea.style.height = "auto";

        textarea.style.height =
            Math.min(textarea.scrollHeight, 180) + "px";
    }

    function toggleSidebar() {
        $("sidebar").classList.toggle("open");
    }


    /*
     * Initialization
     */

    async function initialize() {
        await loadKeyStatus();
        await loadModels();
        await refreshConversations();
    }

    document.addEventListener("DOMContentLoaded", initialize);
