@app.route("/conversations", methods=["POST"])
def new_conversation():
    try:
        conversation_id = create_conversation()

        with get_db() as db:
            row = db.execute(
                """
                SELECT id, title, created_at, updated_at
                FROM conversations
                WHERE id = ?
                """,
                (conversation_id,)
            ).fetchone()

        if row is None:
            app.logger.error(
                "Conversation %s was created but could not be retrieved.",
                conversation_id
            )

            return jsonify({
                "success": False,
                "error": "تم إنشاء المحادثة، لكن تعذر استرجاع بياناتها."
            }), 500

        conversation = dict(row)

        return jsonify({
            "success": True,
            "conversation": conversation,
            "id": conversation_id
        }), 201

    except Exception:
        app.logger.exception(
            "Error while creating a conversation"
        )

        return jsonify({
            "success": False,
            "error": "حدث خطأ أثناء إنشاء المحادثة. راجع نافذة CMD."
        }), 500
