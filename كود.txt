import os
import json
import sqlite3
import requests

from flask import Flask, request, jsonify, render_template_string

app = Flask(__name__)

BASE_DIR = os.path.dirname(os.path.abspath(__file__))
DB_PATH = os.path.join(BASE_DIR, "ai_assistant.db")

PROVIDERS = {
    "openai": {
        "name": "ChatGPT",
        "key_url": "https://platform.openai.com/api-keys",
    },
    "anthropic": {
        "name": "Claude",
        "key_url": "https://console.anthropic.com/settings/keys",
    },
    "gemini": {
        "name": "Gemini",
        "key_url": "https://aistudio.google.com/apikey",
    },
}


def db():
    conn = sqlite3.connect(DB_PATH)
    conn.row_factory = sqlite3.Row
    return conn


def init_db():
    with db() as conn:
        conn.execute("""
            CREATE TABLE IF NOT EXISTS settings (
                provider TEXT PRIMARY KEY,
                api_key TEXT NOT NULL DEFAULT ''
            )
        """)
        conn.execute("""
            CREATE TABLE IF NOT EXISTS chats (
                id INTEGER PRIMARY KEY AUTOINCREMENT,
                title TEXT NOT NULL,
                provider TEXT NOT NULL,
                model TEXT NOT NULL,
                messages TEXT NOT NULL DEFAULT '[]',
                created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
            )
        """)

        for provider in PROVIDERS:
            conn.execute(
                "INSERT OR IGNORE INTO settings(provider, api_key) VALUES (?, '')",
                (provider,)
            )


def get_key(provider):
    with db() as conn:
        row = conn.execute(
            "SELECT api_key FROM settings WHERE provider=?",
            (provider,)
        ).fetchone()
    return row["api_key"] if row else ""


def get_keys():
    return {p: get_key(p) for p in PROVIDERS}


def fetch_models(provider):
    key = get_key(provider)
    if not key:
        return []

    try:
        if provider == "openai":
            r = requests.get(
                "https://api.openai.com/v1/models",
                headers={"Authorization": f"Bearer {key}"},
                timeout=20
            )
            r.raise_for_status()
            data = r.json().get("data", [])

            return sorted(
                [
                    {
                        "id": m["id"],
                        "name": m["id"],
                        "provider": provider
                    }
                    for m in data
                    if m.get("id", "").startswith(
                        ("gpt-", "o1", "o3", "o4")
                    )
                ],
                key=lambda x: x["id"]
            )

        elif provider == "anthropic":
            r = requests.get(
                "https://api.anthropic.com/v1/models",
                headers={
                    "x-api-key": key,
                    "anthropic-version": "2023-06-01"
                },
                params={"limit": 100},
                timeout=20
            )
            r.raise_for_status()
            data = r.json().get("data", [])

            return [
                {
                    "id": m["id"],
                    "name": m.get("display_name", m["id"]),
                    "provider": provider
                }
                for m in data
            ]

        elif provider == "gemini":
            r = requests.get(
                "https://generativelanguage.googleapis.com/v1beta/models",
                params={"key": key, "pageSize": 1000},
                timeout=20
            )
            r.raise_for_status()
            data = r.json().get("models", [])

            return [
                {
                    "id": m["name"].replace("models/", ""),
                    "name": m.get("displayName", m["name"]),
                    "provider": provider
                }
                for m in data
                if "generateContent" in m.get("supportedGenerationMethods", [])
            ]

    except requests.RequestException as e:
        print(f"Model loading error ({provider}): {e}")

    return []


def call_ai(provider, model, messages):
    key = get_key(provider)
    if not key:
        raise ValueError(f"API key for {provider} is missing.")

    if provider == "openai":
        response = requests.post(
            "https://api.openai.com/v1/chat/completions",
            headers={
                "Authorization": f"Bearer {key}",
                "Content-Type": "application/json"
            },
            json={
                "model": model,
                "messages": messages
            },
            timeout=120
        )

        data = response.json()
        if not response.ok:
            raise ValueError(data.get("error", {}).get("message", str(data)))

        return data["choices"][0]["message"]["content"]

    elif provider == "anthropic":
        system_messages = [
            m["content"] for m in messages if m["role"] == "system"
        ]
        chat_messages = [
            {"role": m["role"], "content": m["content"]}
            for m in messages
            if m["role"] in ("user", "assistant")
        ]

        payload = {
            "model": model,
            "max_tokens": 4096,
            "messages": chat_messages
        }

        if system_messages:
            payload["system"] = "\n".join(system_messages)

        response = requests.post(
            "https://api.anthropic.com/v1/messages",
            headers={
                "x-api-key": key,
                "anthropic-version": "2023-06-01",
                "Content-Type": "application/json"
            },
            json=payload,
            timeout=120
        )

        data = response.json()
        if not response.ok:
            raise ValueError(data.get("error", {}).get("message", str(data)))

        return "".join(
            block.get("text", "")
            for block in data.get("content", [])
            if block.get("type") == "text"
        )

    elif provider == "gemini":
        system_messages = [
            m["content"] for m in messages if m["role"] == "system"
        ]
        chat_messages = [
            m for m in messages if m["role"] in ("user", "assistant")
        ]

        contents = []
        for m in chat_messages:
            contents.append({
                "role": "user" if m["role"] == "user" else "model",
                "parts": [{"text": m["content"]}]
            })

        payload = {"contents": contents}

        if system_messages:
            payload["systemInstruction"] = {
                "parts": [{"text": "\n".join(system_messages)}]
            }

        response = requests.post(
            f"https://generativelanguage.googleapis.com/v1beta/models/"
            f"{model}:generateContent",
            params={"key": key},
            json=payload,
            timeout=120
        )

        data = response.json()
        if not response.ok:
            raise ValueError(
                data.get("error", {}).get("message", str(data))
            )

        return "".join(
            part.get("text", "")
            for part in data["candidates"][0]["content"]["parts"]
            if "text" in part
        )

    raise ValueError("Unsupported provider.")

@app.route("/api/settings", methods=["GET"])
def settings_get():
    return jsonify({
        provider: {
            "configured": bool(get_key(provider)),
            "name": info["name"],
            "key_url": info["key_url"]
        }
        for provider, info in PROVIDERS.items()
    })


@app.route("/api/settings", methods=["POST"])
def settings_post():
    data = request.get_json(force=True)

    with db() as conn:
        for provider in PROVIDERS:
            if provider in data:
                key = str(data[provider]).strip()
                conn.execute(
                    """INSERT INTO settings(provider, api_key)
                       VALUES (?, ?)
                       ON CONFLICT(provider)
                       DO UPDATE SET api_key=excluded.api_key""",
                    (provider, key)
                )

    return jsonify({"ok": True})


@app.route("/api/models")
def models():
    result = {}
    for provider, info in PROVIDERS.items():
        result[provider] = {
            "name": info["name"],
            "configured": bool(get_key(provider)),
            "models": fetch_models(provider)
        }

    return jsonify(result)


@app.route("/api/chats", methods=["GET"])
def list_chats():
    with db() as conn:
        rows = conn.execute(
            "SELECT id, title, provider, model, created_at "
            "FROM chats ORDER BY id DESC"
        ).fetchall()
    return jsonify([dict(row) for row in rows])


@app.route("/api/chats", methods=["POST"])
def create_chat():
    data = request.get_json(force=True)
    provider = data.get("provider")
    model = data.get("model")

    if provider not in PROVIDERS or not model:
        return jsonify({"error": "Invalid provider or model"}), 400

    with db() as conn:
        cur = conn.execute(
            "INSERT INTO chats(title, provider, model, messages) "
            "VALUES (?, ?, ?, '[]')",
            ("New conversation", provider, model)
        )
        chat_id = cur.lastrowid

    return jsonify({"id": chat_id})


@app.route("/api/chats/<int:chat_id>", methods=["GET"])
def get_chat(chat_id):
    with db() as conn:
        row = conn.execute(
            "SELECT * FROM chats WHERE id=?", (chat_id,)
        ).fetchone()

    if not row:
        return jsonify({"error": "Chat not found"}), 404

    return jsonify({
        "id": row["id"],
        "title": row["title"],
        "provider": row["provider"],
        "model": row["model"],
        "messages": json.loads(row["messages"])
    })


@app.route("/api/chats/<int:chat_id>/send", methods=["POST"])
def send_message(chat_id):
    data = request.get_json(force=True)
    content = str(data.get("message", "")).strip()

    if not content:
        return jsonify({"error": "Message is empty"}), 400

    with db() as conn:
        row = conn.execute(
            "SELECT * FROM chats WHERE id=?", (chat_id,)
        ).fetchone()

    if not row:
        return jsonify({"error": "Chat not found"}), 404

    messages = json.loads(row["messages"])
    messages.append({"role": "user", "content": content})

    try:
        answer = call_ai(row["provider"], row["model"], messages)
    except Exception as e:
        return jsonify({"error": str(e)}), 400

    messages.append({"role": "assistant", "content": answer})

    title = row["title"]
    if title == "New conversation":
        title = content[:50]

    with db() as conn:
        conn.execute(
            "UPDATE chats SET messages=?, title=? WHERE id=?",
            (json.dumps(messages, ensure_ascii=False), title, chat_id)
        )

    return jsonify({"answer": answer, "messages": messages})

let allModels = {};
let currentChat = null;

async function api(url, options={}) {
    const response = await fetch(url, {
        headers: {'Content-Type':'application/json'},
        ...options
    });
    const data = await response.json();
    if (!response.ok) throw new Error(data.error || 'Request failed');
    return data;
}

async function loadSettings() {
    const data = await api('/api/settings');
    for (const p of ['openai','anthropic','gemini']) {
        document.getElementById(p).placeholder =
            data[p].configured ? 'Key saved — enter new key to change' : 'API key not set';
    }
}

async function saveKeys() {
    const data = {};
    for (const p of ['openai','anthropic','gemini']) {
        const value = document.getElementById(p).value.trim();
        if (value) data[p] = value;
    }
    try {
        await api('/api/settings', {
            method:'POST', body:JSON.stringify(data)
        });
        for (const p of ['openai','anthropic','gemini'])
            document.getElementById(p).value = '';
        document.getElementById('status').textContent =
            'Keys saved. Refreshing model lists...';
        await loadSettings();
        await loadModels();
    } catch(e) {
        document.getElementById('status').textContent = e.message;
    }
}

async function loadModels() {
    const status = document.getElementById('status');
    status.textContent = 'Loading models...';
    try {
        allModels = await api('/api/models');
        updateModels();
        let lines = [];
        for (const [p, group] of Object.entries(allModels)) {
            lines.push(group.name + ': ' +
                (group.configured
                    ? group.models.length + ' models found'
                    : 'API key not set'));
        }
        status.textContent = lines.join('\\n');
    } catch(e) {
        status.textContent = e.message;
    }
}

function updateModels() {
    const provider = document.getElementById('provider').value;
    const select = document.getElementById('model');
    select.innerHTML = '';
    const models = (allModels[provider] || {}).models || [];
    for (const m of models) {
        const option = document.createElement('option');
        option.value = m.id;
        option.textContent = m.name + ' (' + m.id + ')';
        select.appendChild(option);
    }
    if (!models.length) {
        const option = document.createElement('option');
        option.textContent = 'No models loaded — save key and refresh';
        option.value = '';
        select.appendChild(option);
    }
}

async function newChat() {
    const provider = document.getElementById('provider').value;
    const model = document.getElementById('model').value;
    if (!model) {
        alert('Select a model first. Check your API key and refresh models.');
        return;
    }
    try {
        const data = await api('/api/chats', {
            method:'POST',
            body:JSON.stringify({provider, model})
        });
        currentChat = data.id;
        document.getElementById('messages').innerHTML = '';
        await loadChats();
    } catch(e) { alert(e.message); }
}

function renderMessages(messages) {
    const container = document.getElementById('messages');
    container.innerHTML = '';
    for (const m of messages) {
        const div = document.createElement('div');
        div.className = 'msg ' + m.role;
        div.textContent = (m.role === 'user' ? 'You' : 'AI') +
            '\\n\\n' + m.content;
        container.appendChild(div);
    }
    container.scrollTop = container.scrollHeight;
}

async function openChat(id) {
    try {
        const chat = await api('/api/chats/' + id);
        currentChat = id;
        document.getElementById('provider').value = chat.provider;
        await loadModels();
        document.getElementById('provider').value = chat.provider;
        updateModels();
        document.getElementById('model').value = chat.model;
        renderMessages(chat.messages);
    } catch(e) { alert(e.message); }
}

async function loadChats() {
    const chats = await api('/api/chats');
    const list = document.getElementById('chatList');
    list.innerHTML = '';
    for (const chat of chats) {
        const button = document.createElement('button');
        button.textContent = chat.title + ' [' + chat.provider + ']';
        button.onclick = () => openChat(chat.id);
        list.appendChild(button);
    }
}

async function sendMessage() {
    const input = document.getElementById('message');
    const message = input.value.trim();

    if (!currentChat) await newChat();

        await loadChats();
    } catch(e) {
        pending.textContent = 'Error: ' + e.message;
    }
}
