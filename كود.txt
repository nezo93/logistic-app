from flask import Flask, render_template, request, jsonify
from google import genai
import os

app = Flask(__name__)

API_KEY = "YOUR_API_KEY"

client = genai.Client(
    api_key=API_KEY,
    http_options={"timeout": 60000}
)

MODELS = {
    "gemini-3.8-flash": "Gemini 3.8 Flash",
    "gemini-3.7-flash": "Gemini 3.7 Flash",
    "gemini-3.6-flash": "Gemini 3.6 Flash",
    "gemini-3.5-flash": "Gemini 3.5 Flash",
    "gemini-3.5-flash-lite": "Gemini 3.5 Flash-Lite",
    "gemini-3.1-flash-lite": "Gemini 3.1 Flash-Lite"
}


@app.route("/")
def home():
    return render_template("index.html")


@app.route("/chat", methods=["POST"])
def chat():
    data = request.get_json(silent=True) or {}

    question = data.get("message", "").strip()
    selected_model = data.get("model", "gemini-3.8-flash")
    history = data.get("history", [])

    if not question:
        return jsonify({"error": "Please enter a message."}), 400

    if selected_model != "auto" and selected_model not in MODELS:
        return jsonify({"error": "Invalid model selection."}), 400

    if selected_model == "auto":
        models_to_try = list(MODELS.keys())
    else:
        models_to_try = [selected_model]

    # Build the conversation input
    conversation = []

    if isinstance(history, list):
        for item in history[-20:]:
            if not isinstance(item, dict):
                continue

            role = item.get("role")
            content = item.get("content")

            if role in ("user", "assistant") and isinstance(content, str):
                conversation.append(f"{role}: {content}")

    conversation.append(f"user: {question}")

    prompt = "\n\n".join(conversation)

    last_error = "No model was available."

    for model_id in models_to_try:
        try:
            response = client.interactions.create(
                model=model_id,
                input=prompt
            )

            answer = response.output_text

            if not answer:
                answer = "The model returned an empty response."

            return jsonify({
                "reply": answer,
                "model": MODELS[model_id]
            })

        except Exception as error:
            last_error = f"{type(error).__name__}: {error}"

            if selected_model != "auto":
                break

    print("Gemini API error:", last_error)

    return jsonify({
        "error": (
            "The AI request failed. Check the model availability, "
            "API key, internet connection, and account limits."
        )
    }), 502


if __name__ == "__main__":
    app.run(
        host="127.0.0.1",
        port=5000,
        debug=False
    )
