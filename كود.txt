                        Enter للإرسال · Shift + Enter لسطر جديد

                            ⏹ إيقاف

                            إرسال ↑
                        </button>

    <script>
        let currentConversationId = null;
        let isGenerating = false;
        let activeController = null;
        let activeRequestId = null;
        let stopRequested = false;

        const messagesElement = document.getElementById("messages");
        const inputElement = document.getElementById("messageInput");
        const sendButton = document.getElementById("sendButton");
        const stopButton = document.getElementById("stopButton");
        const statusElement = document.getElementById("status");

        function addMessage(role, content, model = "") {
            const welcome = document.getElementById("welcome");
            if (welcome) welcome.remove();

            const box = document.createElement("div");
            box.className = "message " + role;

            const roleLabel = document.createElement("div");
            roleLabel.className = "role";
            roleLabel.textContent =
                role === "user" ? "أنت" :
                "المساعد" + (model ? " · " + model : "");

            const text = document.createElement("div");
            text.textContent = content;

            box.appendChild(roleLabel);
            box.appendChild(text);
            messagesElement.appendChild(box);

            messagesElement.scrollTop = messagesElement.scrollHeight;

            return box;
        }

        function showWelcome() {
            messagesElement.innerHTML = `
                <div class="welcome" id="welcome">
                    <h1>كيف يمكنني مساعدتك؟</h1>
                    <p>اكتب سؤالك للبدء.</p>
                </div>
            `;
        }

        async function refreshConversations() {
            try {
                const response = await fetch("/conversations");
                const conversations = await response.json();

                const list = document.getElementById("conversationList");
                list.innerHTML = "";

                conversations.forEach(conversation => {
                    const item = document.createElement("div");
                    item.className = "conversation";

                    if (conversation.id === currentConversationId) {
                        item.classList.add("active");
                    }

                    item.textContent = conversation.title;
                    item.title = conversation.title;

                    item.onclick = () => loadConversation(conversation.id);

                    list.appendChild(item);
                });
            } catch (error) {
                statusElement.textContent =
                    "تعذر تحميل قائمة المحادثات.";
            }
        }

        async function newChat() {
            if (isGenerating) {
                statusElement.textContent =
                    "أوقف الإجابة الحالية أولاً قبل إنشاء محادثة جديدة.";
                return;
            }

            try {
                const response = await fetch("/conversations", {
                    method: "POST"
                });

                const data = await response.json();
                currentConversationId = data.id;

                showWelcome();
                statusElement.textContent = "";
                await refreshConversations();
                inputElement.focus();
            } catch (error) {
                statusElement.textContent =
                    "تعذر إنشاء محادثة جديدة.";
            }
        }

        async function loadConversation(id) {
            if (isGenerating) {
                statusElement.textContent =
                    "أوقف الإجابة الحالية أولاً.";
                return;
            }

            try {
                const response = await fetch(
                    "/conversations/" + encodeURIComponent(id)
                );

                const data = await response.json();

                if (!response.ok) {
                    throw new Error(data.error || "فشل تحميل المحادثة");
                }

                currentConversationId = data.id;
                messagesElement.innerHTML = "";

                if (!data.messages.length) {
                    showWelcome();
                } else {
                    data.messages.forEach(message => {
                        addMessage(
                            message.role === "assistant" ? "assistant" : "user",
                            message.content,
                            message.model || ""
                        );
                    });
                }

                statusElement.textContent = "";
                await refreshConversations();
            } catch (error) {
                statusElement.textContent =
                    "تعذر تحميل المحادثة: " + error.message;
            }
        }

        async function sendMessage() {
            const message = inputElement.value.trim();

            if (!message || isGenerating) return;

            if (!currentConversationId) {
                const response = await fetch("/conversations", {
                    method: "POST"
                });

                const data = await response.json();
                currentConversationId = data.id;
            }

            const model = document.getElementById("model").value;

            addMessage("user", message);
            inputElement.value = "";

            isGenerating = true;
            stopRequested = false;
            activeController = new AbortController();
            activeRequestId = null;

            sendButton.disabled = true;
            stopButton.style.display = "inline-block";
            statusElement.textContent = "جارٍ التفكير...";

            // مؤشر بصري للإجابة قيد الانتظار
            const pendingBox = addMessage("assistant", "جارٍ إنشاء الإجابة...");

            try {
                const response = await fetch("/chat", {
                    method: "POST",
                    headers: {
                        "Content-Type": "application/json"
                    },
                    signal: activeController.signal,
                    body: JSON.stringify({
                        message: message,
                        model: model,
                        conversation_id: currentConversationId
                    })
                });

                const data = await response.json();

                if (stopRequested) {
                    pendingBox.querySelector("div:last-child").textContent =
                        "تم إيقاف الانتظار. قد يستمر الطلب على الخادم.";
                    return;
                }

                if (!response.ok || data.error) {
                    pendingBox.querySelector("div:last-child").textContent =
                        "حدث خطأ: " + (data.error || "تعذر الحصول على الإجابة");
                    return;
                }

                currentConversationId = data.conversation_id;
                pendingBox.querySelector("div:last-child").textContent =
                    data.reply || "لم يصل رد نصي.";

                pendingBox.querySelector(".role").textContent =
                    "المساعد · " + (data.model || model);

                statusElement.textContent = "";
            } catch (error) {
                if (stopRequested || error.name === "AbortError") {
                    pendingBox.querySelector("div:last-child").textContent =
                        "تم إيقاف انتظار الإجابة.";
                } else {
                    pendingBox.querySelector("div:last-child").textContent =
                        "تعذر الاتصال: " + error.message;
                }
            } finally {
                isGenerating = false;
                activeController = null;
                activeRequestId = null;
                sendButton.disabled = false;
                stopButton.style.display = "none";

                if (!stopRequested) {
                    statusElement.textContent = "";
                }

                await refreshConversations();
            }
        }

        async function stopGeneration() {
            if (!isGenerating) return;

            stopRequested = true;
            statusElement.textContent = "جارٍ إيقاف الانتظار...";

            // إلغاء انتظار المتصفح فوراً.
            if (activeController) {
                activeController.abort();
            }

            // طلب إلغاء من الخادم إذا توفر معرّف الطلب.
            if (activeRequestId) {
                try {
                    await fetch("/stop", {
                        method: "POST",
                        headers: {
                            "Content-Type": "application/json"
                        },
                        body: JSON.stringify({
                            request_id: activeRequestId
                        })
                    });
                } catch (error) {
                    // ربما أُلغي اتصال المتصفح بالفعل.
                }
            }

            statusElement.textContent =
                "أُوقف انتظار الإجابة. قد تستمر المعالجة على خوادم المزود.";
        }

        function handleKey(event) {
            if (event.key === "Enter" && !event.shiftKey) {
                event.preventDefault();
                sendMessage();
            }
        }

        async function initialize() {
            await refreshConversations();

            const response = await fetch("/conversations");
            const conversations = await response.json();

            if (conversations.length) {
                await loadConversation(conversations[0].id);
            } else {
                await newChat();
            }
        }

        initialize();
    </script>
