from flask import Flask, render_template, request, jsonify
from google import genai
import sqlite3
import os
import threading
import uuid
from datetime import datetime

app = Flask(__name__)

# ضع مفتاح Gemini هنا أو استخدم متغير البيئة GEMINI_API_KEY
API_KEY = os.environ.get("GEMINI_API_KEY", "YOUR_API_KEY")
client = genai.Client(api_key=API_KEY)

DB_FILE = "assistant.db"

# معرفات النماذج: قد تختلف الإتاحة حسب حسابك وواجهة API.
MODELS = [
    "gemini-3.8-flash",
    "gemini-3.7-flash",
    "gemini-3.6-flash",
    "gemini-3.5-flash",
    "gemini-3.5-flash-lite",
    "gemini-3.1-flash-lite",
]

# الطلبات النشطة التي يمكن محاولة إلغائها
active_requests = {}
requests_lock = threading.Lock()


def get_db():
    conn = sqlite3.connect(DB_FILE, timeout=30)
    conn.row_factory = sqlite3.Row
    return conn


def init_db():
    with get_db() as conn:
        conn.execute("""
            CREATE TABLE IF NOT EXISTS conversations (
                id TEXT PRIMARY KEY,
                title TEXT NOT NULL,
                created_at TEXT NOT NULL,
                updated_at TEXT NOT NULL
            )
        """)

        conn.execute("""
            CREATE TABLE IF NOT EXISTS messages (
                id INTEGER PRIMARY KEY AUTOINCREMENT,
                conversation_id TEXT NOT NULL,
                role TEXT NOT NULL,
                content TEXT NOT NULL,
                model TEXT,
                created_at TEXT NOT NULL,
                FOREIGN KEY (conversation_id)
                    REFERENCES conversations(id)
            )
        """)


def now():
    return datetime.now().isoformat(timespec="seconds")


def save_message(conversation_id, role, content, model=None):
    timestamp = now()

    with get_db() as conn:
        conn.execute("""
            INSERT INTO messages
                (conversation_id, role, content, model, created_at)
            VALUES (?, ?, ?, ?, ?)
        """, (conversation_id, role, content, model, timestamp))

        conn.execute("""
            UPDATE conversations
            SET updated_at = ?
            WHERE id = ?
        """, (timestamp, conversation_id))


def conversation_exists(conversation_id):
    with get_db() as conn:
        row = conn.execute(
            "SELECT id FROM conversations WHERE id = ?",
            (conversation_id,)
        ).fetchone()
    return row is not None


def create_conversation(title="محادثة جديدة"):
    conversation_id = str(uuid.uuid4())
    timestamp = now()

    with get_db() as conn:
        conn.execute("""
            INSERT INTO conversations
                (id, title, created_at, updated_at)
            VALUES (?, ?, ?, ?)
        """, (conversation_id, title, timestamp, timestamp))

    return conversation_id


def get_messages(conversation_id):
    with get_db() as conn:
        rows = conn.execute("""
            SELECT role, content, model, created_at
            FROM messages
            WHERE conversation_id = ?
            ORDER BY id ASC
        """, (conversation_id,)).fetchall()

    return [dict(row) for row in rows]


@app.route("/")
def index():
    return render_template("index.html", models=MODELS)


@app.route("/conversations", methods=["GET"])
def list_conversations():
    with get_db() as conn:
        rows = conn.execute("""
            SELECT id, title, created_at, updated_at
            FROM conversations
            ORDER BY updated_at DESC
        """).fetchall()

    return jsonify([dict(row) for row in rows])


@app.route("/conversations", methods=["POST"])
def new_conversation():
    conversation_id = create_conversation()

    return jsonify({
        "id": conversation_id,
        "title": "محادثة جديدة"
    })


@app.route("/conversations/<conversation_id>", methods=["GET"])
def load_conversation(conversation_id):
    if not conversation_exists(conversation_id):
        return jsonify({"error": "المحادثة غير موجودة"}), 404

    return jsonify({
        "id": conversation_id,
        "messages": get_messages(conversation_id)
    })


@app.route("/conversations/<conversation_id>", methods=["DELETE"])
def delete_conversation(conversation_id):
    with get_db() as conn:
        conn.execute(
            "DELETE FROM messages WHERE conversation_id = ?",
            (conversation_id,)
        )
        conn.execute(
            "DELETE FROM conversations WHERE id = ?",
            (conversation_id,)
        )

    return jsonify({"success": True})


@app.route("/chat", methods=["POST"])
def chat():
    data = request.get_json(silent=True) or {}

    message = str(data.get("message", "")).strip()
    model = data.get("model", MODELS[0])
    conversation_id = data.get("conversation_id")

    if not message:
        return jsonify({"error": "اكتب رسالة أولاً"}), 400

    if model not in MODELS:
        return jsonify({"error": "النموذج المختار غير موجود في القائمة"}), 400

    if not conversation_id or not conversation_exists(conversation_id):
        conversation_id = create_conversation(
            message[:60] if message else "محادثة جديدة"
        )

    # حفظ رسالة المستخدم فوراً
    save_message(conversation_id, "user", message, model)

    # حفظ عنوان المحادثة من أول رسالة
    with get_db() as conn:
        row = conn.execute("""
            SELECT COUNT(*) AS count
            FROM messages
            WHERE conversation_id = ?
        """, (conversation_id,)).fetchone()

        if row["count"] == 1:
            conn.execute("""
                UPDATE conversations
                SET title = ?
                WHERE id = ?
            """, (message[:60], conversation_id))

    # إنشاء طلب يمكن للواجهة طلب إلغائه
    request_id = str(uuid.uuid4())
    cancel_event = threading.Event()

    with requests_lock:
        active_requests[request_id] = cancel_event

    try:
        # بناء السياق من المحادثة المحفوظة.
        # استبعاد آخر رسالة من السجل لتجنب تكرارها.
        history = get_messages(conversation_id)[:-1]

        prompt_parts = [
            "أنت مساعد ذكاء اصطناعي مفيد. أجب بلغة المستخدم.",
            "هذه رسائل المحادثة السابقة:"
        ]

        for item in history:
            speaker = "المستخدم" if item["role"] == "user" else "المساعد"
            prompt_parts.append(f"{speaker}: {item['content']}")

        prompt_parts.append(f"المستخدم: {message}")
        prompt_parts.append("المساعد:")

        prompt = "\n\n".join(prompt_parts)

        if cancel_event.is_set():
            return jsonify({
                "cancelled": True,
                "conversation_id": conversation_id
            })

        # واجهة Gemini المتزامنة؛ إلغاء الانتظار لا يضمن إيقاف
        # المعالجة داخل خوادم المزود.
        response = client.models.generate_content(
            model=model,
            contents=prompt
        )

        if cancel_event.is_set():
            return jsonify({
                "cancelled": True,
                "conversation_id": conversation_id
            })

        reply = getattr(response, "text", None) or "لم يصل رد نصي من النموذج."

        # حفظ الرد بعد اكتماله
        save_message(conversation_id, "assistant", reply, model)

        return jsonify({
            "reply": reply,
            "model": model,
            "conversation_id": conversation_id
        })

    except Exception as e:
        return jsonify({
            "error": str(e),
            "conversation_id": conversation_id
        }), 500

    finally:
        with requests_lock:
            active_requests.pop(request_id, None)


@app.route("/stop", methods=["POST"])
def stop_request():
    data = request.get_json(silent=True) or {}
    request_id = data.get("request_id")

    with requests_lock:
        event = active_requests.get(request_id)

        if event:
            event.set()
            return jsonify({
                "success": True,
                "message": "تم طلب إيقاف الإجابة"
            })

    return jsonify({
        "success": False,
        "message": "لم يتم العثور على طلب نشط"
    })


init_db()

if __name__ == "__main__":
    app.run(debug=False, threaded=True)
