# =========================================================
# General helpers
# =========================================================

def now_iso():
    return datetime.now(timezone.utc).isoformat()


def get_db():
    db = sqlite3.connect(
        DATABASE_PATH,
        timeout=30
    )

    db.row_factory = sqlite3.Row

    db.execute("PRAGMA foreign_keys = ON")
    db.execute("PRAGMA busy_timeout = 30000")

    return db


def row_to_dict(row):
    return dict(row) if row else None


def valid_provider(provider):
    return provider in PROVIDERS


def error_response(message, status=400, **extra):
    data = {
        "error": str(message)
    }

    data.update(extra)

    return jsonify(data), status


# =========================================================
# Database initialization and migration
# =========================================================

def init_db():
    with get_db() as db:

        db.execute("""
            CREATE TABLE IF NOT EXISTS conversations (
                id INTEGER PRIMARY KEY AUTOINCREMENT,
                title TEXT NOT NULL DEFAULT 'محادثة جديدة',
                created_at TEXT NOT NULL,
                updated_at TEXT NOT NULL
            )
        """)

        db.execute("""
            CREATE TABLE IF NOT EXISTS messages (
                id INTEGER PRIMARY KEY AUTOINCREMENT,
                conversation_id INTEGER NOT NULL,
                role TEXT NOT NULL,
                content TEXT NOT NULL,
                provider TEXT,
                model TEXT,
                created_at TEXT NOT NULL,
                FOREIGN KEY (conversation_id)
                    REFERENCES conversations(id)
                    ON DELETE CASCADE
            )
        """)

        # Migrate older database versions safely.
        columns = {
            row["name"]
            for row in db.execute(
                "PRAGMA table_info(messages)"
            ).fetchall()
        }

        if "provider" not in columns:
            db.execute("""
                ALTER TABLE messages
                ADD COLUMN provider TEXT
            """)

        if "model" not in columns:
            db.execute("""
                ALTER TABLE messages
                ADD COLUMN model TEXT
            """)

        if "created_at" not in columns:
            db.execute("""
                ALTER TABLE messages
                ADD COLUMN created_at TEXT
            """)

            db.execute("""
                UPDATE messages
                SET created_at = ?
                WHERE created_at IS NULL
            """, (now_iso(),))

        db.execute("""
            CREATE INDEX IF NOT EXISTS
            idx_messages_conversation
            ON messages(conversation_id, id)
        """)

        db.execute("""
            CREATE INDEX IF NOT EXISTS
            idx_conversations_updated
            ON conversations(updated_at)
        """)

        db.commit()


init_db()

def read_api_keys():
    if not os.path.exists(API_KEYS_PATH):
        return {}

    try:
        with open(
            API_KEYS_PATH,
            "r",
            encoding="utf-8"
        ) as file:
            data = json.load(file)

        if not isinstance(data, dict):
            return {}

        return data

    except (json.JSONDecodeError, OSError):
        return {}


def get_api_key(provider):
    keys = read_api_keys()

    value = keys.get(provider, "")

    # Support older files that stored objects per provider.
    if isinstance(value, dict):
        value = value.get("api_key", value.get("key", ""))

    if not isinstance(value, str):
        return ""

    return value.strip()


def save_api_key(provider, api_key):
    if not valid_provider(provider):
        raise ValueError("مزود الخدمة غير معروف.")

    keys = read_api_keys()
    keys[provider] = api_key.strip()

    temporary_path = API_KEYS_PATH + ".tmp"

    with open(
        temporary_path,
        "w",
        encoding="utf-8"
    ) as file:
        json.dump(
            keys,
            file,
            ensure_ascii=False,
            indent=2
        )

    os.replace(temporary_path, API_KEYS_PATH)


@app.route("/api/keys", methods=["GET"])
def list_api_keys():
    """
    Return only whether each provider has a key.
    Never expose the actual API keys to the browser.
    """

    keys = {}

    for provider in PROVIDERS:
        keys[provider] = bool(get_api_key(provider))

    return jsonify({
        "keys": keys
    })


@app.route("/api/keys", methods=["POST"])
def update_api_key():
    data = request.get_json(silent=True) or {}

    provider = data.get("provider")
    api_key = data.get("api_key")

    if not valid_provider(provider):
        return error_response(
            "مزود الخدمة غير معروف."
        )

    if not isinstance(api_key, str):
        return error_response(
            "مفتاح API غير صالح."
        )

    try:
        save_api_key(provider, api_key)

        return jsonify({
            "success": True,
            "provider": provider,
            "configured": bool(api_key.strip())
        })

    except OSError:
        app.logger.exception("Failed to save API key")

        return error_response(
            "تعذر حفظ مفتاح API. تحقق من صلاحيات المجلد.",
            500
        )

def provider_error_message(response):
    try:
        data = response.json()

        if isinstance(data, dict):
            error = data.get("error", {})

            if isinstance(error, dict):
                return (
                    error.get("message")
                    or error.get("code")
                    or str(error)
                )

            if error:
                return str(error)

            return (
                data.get("message")
                or data.get("detail")
                or str(data)
            )

    except ValueError:
        pass

    return response.text[:500] or "خطأ غير معروف من مزود الخدمة."


def get_provider_models(provider):
    api_key = get_api_key(provider)

    if not api_key:
        raise ValueError(
            "لم يتم حفظ مفتاح API لهذا المزود. "
            "أضف المفتاح من إعدادات التطبيق."
        )

    config = PROVIDERS[provider]
    base_url = config["base_url"]

    if config["type"] == "gemini":

        response = requests.get(
            f"{base_url}/models",
            params={
                "key": api_key,
                "pageSize": 100
            },
            timeout=REQUEST_TIMEOUT
        )

        if not response.ok:
            raise RuntimeError(
                provider_error_message(response)
            )

        data = response.json()
        models = []

        for item in data.get("models", []):
            name = item.get("name", "")
            supported_methods = item.get(
                "supportedGenerationMethods", []
            )

            if (
                name.startswith("models/")
                and "generateContent" in supported_methods
            ):
                models.append({
                    "id": name.replace("models/", "", 1),
                    "name": name.replace("models/", "", 1)
                })

        return models

    response = requests.get(
        f"{base_url}/models",
        headers={
            "Authorization": f"Bearer {api_key}"
        },
        timeout=REQUEST_TIMEOUT
    )

    if not response.ok:
        raise RuntimeError(
            provider_error_message(response)
        )

    data = response.json()
    models = []

    for item in data.get("data", []):
        model_id = item.get("id")

        if model_id:
            models.append({
                "id": model_id,
                "name": model_id
            })

    return models

    except Exception as error:
        app.logger.exception(
            "Failed to load models for %s",
            provider
        )

        return error_response(
            str(error),
            502
        )


def call_openai_compatible(
    provider,
    model,
    messages,
    temperature=0.7
):
    api_key = get_api_key(provider)

    if not api_key:
        raise ValueError(
            f"لم يتم حفظ مفتاح API لمزود {PROVIDERS[provider]['name']}."
        )

    base_url = PROVIDERS[provider]["base_url"]

    payload = {
        "model": model,
        "messages": messages,
        "temperature": temperature
    }

    response = requests.post(
        f"{base_url}/chat/completions",
        headers={
            "Authorization": f"Bearer {api_key}",
            "Content-Type": "application/json"
        },
        json=payload,
        timeout=REQUEST_TIMEOUT
    )

    if not response.ok:
        raise RuntimeError(
            provider_error_message(response)
        )

    data = response.json()

    choices = data.get("choices", [])

    if not choices:
        raise RuntimeError(
            "لم يرجع مزود الخدمة أي إجابة."
        )

    message = choices[0].get("message", {})
    content = message.get("content", "")

    if isinstance(content, list):
        text_parts = []

        for part in content:
            if isinstance(part, dict):
                if part.get("type") == "text":
                    text_parts.append(
                        part.get("text", "")
                    )

        content = "\n".join(text_parts)

    if not isinstance(content, str) or not content.strip():
        raise RuntimeError(
            "أعاد النموذج إجابة فارغة."
        )

    return content.strip()


def call_gemini(model, messages, temperature=0.7):
    api_key = get_api_key("gemini")

    if not api_key:
        raise ValueError(
            "لم يتم حفظ مفتاح Gemini API."
        )

    contents = []
    system_instruction = None

    for message in messages:
        role = message.get("role")
        content = message.get("content", "")

        if role == "system":
            system_instruction = {
                "parts": [{
                    "text": content
                }]
            }

        elif role in ("user", "assistant"):
            contents.append({
                "role": (
                    "model"
                    if role == "assistant"
                    else "user"
                ),
                "parts": [{
                    "text": content
                }]
            })

    payload = {
        "contents": contents,
        "generationConfig": {
            "temperature": temperature
        }
    }

    if system_instruction:
        payload["systemInstruction"] = system_instruction

    url = (
        f"{PROVIDERS['gemini']['base_url']}"
        f"/models/{model}:generateContent"
    )

    response = requests.post(
        url,
        params={
            "key": api_key
        },
        json=payload,
        timeout=REQUEST_TIMEOUT
    )

    if not response.ok:
        raise RuntimeError(
            provider_error_message(response)
        )

    data = response.json()

    candidates = data.get("candidates", [])

    if not candidates:
        feedback = data.get("promptFeedback", {})

        raise RuntimeError(
            "لم يُرجع Gemini إجابة. "
            + str(feedback or data)
        )

    parts = (
        candidates[0]
        .get("content", {})
        .get("parts", [])
    )

    answer = "\n".join(
        part.get("text", "")
        for part in parts
        if isinstance(part, dict)
        and part.get("text")
    )

    if not answer.strip():
        raise RuntimeError(
            "أعاد Gemini إجابة فارغة."
        )

    return answer.strip()


def generate_answer(provider, model, messages):
    if not valid_provider(provider):
        raise ValueError(
            "مزود الخدمة غير معروف."
        )

    if not isinstance(model, str) or not model.strip():
        raise ValueError(
            "اختر نموذجًا قبل إرسال الرسالة."
        )

    if provider == "gemini":
        return call_gemini(
            model.strip(),
            messages
        )

    return call_openai_compatible(
        provider,
        model.strip(),
        messages
    )

def create_conversation(title="محادثة جديدة"):
    timestamp = now_iso()

    with get_db() as db:
        cursor = db.execute("""
            INSERT INTO conversations (
                title,
                created_at,
                updated_at
            )
            VALUES (?, ?, ?)
        """, (
            title,
            timestamp,
            timestamp
        ))

        conversation_id = cursor.lastrowid
        db.commit()

    return conversation_id


def conversation_exists(conversation_id):
    with get_db() as db:
        row = db.execute("""
            SELECT id
            FROM conversations
            WHERE id = ?
        """, (conversation_id,)).fetchone()

    return row is not None


def save_message(
    conversation_id,
    role,
    content,
    provider=None,
    model=None
):
    timestamp = now_iso()

    with get_db() as db:
        db.execute("""
            INSERT INTO messages (
                conversation_id,
                role,
                content,
                provider,
                model,
                created_at
            )
            VALUES (?, ?, ?, ?, ?, ?)
        """, (
            conversation_id,
            role,
            content,
            provider,
            model,
            timestamp
        ))

        db.execute("""
            UPDATE conversations
            SET updated_at = ?
            WHERE id = ?
        """, (
            timestamp,
            conversation_id
        ))

        db.commit()


def get_conversation_messages(conversation_id):
    with get_db() as db:
        rows = db.execute("""
            SELECT
                id,
                role,
                content,
                provider,
                model,
                created_at
            FROM messages
            WHERE conversation_id = ?
            ORDER BY id ASC
        """, (conversation_id,)).fetchall()

    return [dict(row) for row in rows]


def build_model_history(conversation_id):
    rows = get_conversation_messages(
        conversation_id
    )

    history = []

    for row in rows:
        if row["role"] not in ("user", "assistant"):
            continue

        history.append({
            "role": row["role"],
            "content": row["content"]
        })

    return history

@app.route("/conversations", methods=["GET"])
def list_conversations():
    with get_db() as db:
        rows = db.execute("""
            SELECT
                c.id,
                c.title,
                c.created_at,
                c.updated_at,
                (
                    SELECT COUNT(*)
                    FROM messages m
                    WHERE m.conversation_id = c.id
                ) AS message_count
            FROM conversations c
            ORDER BY c.updated_at DESC, c.id DESC
        """).fetchall()

    return jsonify({
        "conversations": [
            dict(row) for row in rows
        ]
    })


@app.route("/conversations", methods=["POST"])
def new_conversation():
    conversation_id = create_conversation()

    with get_db() as db:
        row = db.execute("""
            SELECT *
            FROM conversations
            WHERE id = ?
        """, (conversation_id,)).fetchone()

    return jsonify({
        "success": True,
        "conversation": dict(row),
        "id": conversation_id
    }), 201


@app.route("/conversations/<int:conversation_id>", methods=["GET"])
def open_conversation(conversation_id):
    with get_db() as db:
        conversation = db.execute("""
            SELECT *
            FROM conversations
            WHERE id = ?
        """, (conversation_id,)).fetchone()

    if not conversation:
        return error_response(
            "المحادثة غير موجودة أو تم حذفها.",
            404
        )

    messages = get_conversation_messages(
        conversation_id
    )

    return jsonify({
        "conversation": dict(conversation),
        "messages": messages
    })


@app.route(
    "/conversations/<int:conversation_id>",
    methods=["DELETE"]
)
def delete_conversation(conversation_id):
    with get_db() as db:
        cursor = db.execute("""
            DELETE FROM conversations
            WHERE id = ?
        """, (conversation_id,))

        deleted = cursor.rowcount
        db.commit()

    if deleted == 0:
        return error_response(
            "المحادثة غير موجودة أو تم حذفها مسبقًا.",
            404
        )

    return jsonify({
        "success": True,
        "deleted_id": conversation_id
    })

@app.route("/chat", methods=["POST"])
def chat():
    data = request.get_json(silent=True)

    if not isinstance(data, dict):
        return error_response(
            "بيانات الطلب غير صالحة."
        )

    user_message = data.get("message", "")

    if not isinstance(user_message, str):
        return error_response(
            "نص الرسالة غير صالح."
        )

    user_message = user_message.strip()

    if not user_message:
        return error_response(
            "اكتب رسالة أولًا."
        )

    provider = data.get("provider")
    model = data.get("model")

    if not valid_provider(provider):
        return error_response(
            "اختر مزود خدمة صحيحًا."
        )

    if not isinstance(model, str) or not model.strip():
        return error_response(
            "اختر النموذج الذي تريد استخدامه."
        )

    conversation_id = data.get("conversation_id")

    # Create a new conversation if no ID was supplied.
    if conversation_id in (None, "", 0):
        conversation_id = create_conversation()

    else:
        try:
            conversation_id = int(conversation_id)
        except (TypeError, ValueError):
            return error_response(
                "معرّف المحادثة غير صالح."
            )

        if not conversation_exists(conversation_id):
            return error_response(
                "المحادثة غير موجودة. أنشئ محادثة جديدة.",
                404
            )

    # Save the user message before contacting the provider.
    # This allows recovery even if the model request fails.
    try:
        save_message(
            conversation_id,
            "user",
            user_message,
            provider,
            model
        )

    except Exception:
        app.logger.exception(
            "Failed to save user message"
        )

        return error_response(
            "تعذر حفظ الرسالة في قاعدة البيانات.",
            500,
            conversation_id=conversation_id
        )

    # Build history including the just-saved user message.
    try:
        history = build_model_history(
            conversation_id
        )

        answer = generate_answer(
            provider,
            model,
            history
        )

        save_message(
            conversation_id,
            "assistant",
            answer,
            provider,
            model
        )

        with get_db() as db:
            conversation = db.execute("""
                SELECT *
                FROM conversations
                WHERE id = ?
            """, (conversation_id,)).fetchone()

        return jsonify({
            "success": True,
            "conversation_id": conversation_id,
            "conversation": dict(conversation),
            "reply": answer,
            "provider": provider,
            "model": model
        })

    except (ValueError, RuntimeError) as error:
        app.logger.warning(
            "Provider error: %s",
            error
        )

        return error_response(
            str(error),
            502,
            conversation_id=conversation_id
        )

    except requests.RequestException as error:
        app.logger.exception(
            "Network error while generating answer"
        )

        return error_response(
            "تعذر الاتصال بمزود الذكاء الاصطناعي. "
            "تحقق من الإنترنت ثم حاول مرة أخرى.",
            502,
            conversation_id=conversation_id
        )

    except Exception:
        app.logger.exception(
            "Unexpected error in /chat"
        )

        return error_response(
            "حدث خطأ داخلي أثناء معالجة الرسالة. "
            "راجع نافذة CMD لمعرفة التفاصيل.",
            500,
            conversation_id=conversation_id
        )

@app.errorhandler(404)
def not_found(error):
    if request.path.startswith(
        ("/api/", "/chat", "/conversations")
    ):
        return jsonify({
            "error": "المسار المطلوب غير موجود."
        }), 404

@app.errorhandler(500)
def internal_error(error):
    app.logger.exception(
        "Unhandled Flask error"
    )

    if request.path.startswith(
        ("/api/", "/chat", "/conversations")
    ):
        return jsonify({
            "error": "حدث خطأ داخلي في الخادم."
        }), 500

    return "Internal server error", 500
