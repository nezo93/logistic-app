def get_db():
    connection = sqlite3.connect(DB_PATH, timeout=30)
    connection.row_factory = sqlite3.Row

    connection.execute("PRAGMA foreign_keys = ON")

    return connection


def init_db():
    with get_db() as db:

        db.execute("""
            CREATE TABLE IF NOT EXISTS conversations (
                id INTEGER PRIMARY KEY AUTOINCREMENT,
                title TEXT NOT NULL DEFAULT 'محادثة جديدة',
                created_at TEXT NOT NULL,
                updated_at TEXT NOT NULL
            )
        """)

        db.execute("""
            CREATE TABLE IF NOT EXISTS messages (
                id INTEGER PRIMARY KEY AUTOINCREMENT,
                conversation_id INTEGER NOT NULL,
                role TEXT NOT NULL,
                content TEXT NOT NULL,
                provider TEXT,
                model TEXT,
                created_at TEXT NOT NULL,

                FOREIGN KEY (conversation_id)
                    REFERENCES conversations(id)
                    ON DELETE CASCADE
            )
        """)

        db.execute("""
            CREATE INDEX IF NOT EXISTS
            idx_messages_conversation
            ON messages(conversation_id, id)
        """)

        db.commit()


init_db()

def load_keys():
    keys = {
        "gemini": "",
        "openrouter": "",
        "groq": "",
        "deepseek": "",
    }

    if KEYS_PATH.exists():
        try:
            with open(KEYS_PATH, "r", encoding="utf-8") as file:
                saved = json.load(file)

            if isinstance(saved, dict):
                for provider in keys:
                    value = saved.get(provider, "")
                    if isinstance(value, str):
                        keys[provider] = value.strip()

        except (OSError, json.JSONDecodeError):
            pass

    # Environment variables override saved keys.
    environment_names = {
        "gemini": "GEMINI_API_KEY",
        "openrouter": "OPENROUTER_API_KEY",
        "groq": "GROQ_API_KEY",
        "deepseek": "DEEPSEEK_API_KEY",
    }

    for provider, environment_name in environment_names.items():
        environment_value = os.getenv(environment_name, "").strip()

        if environment_value:
            keys[provider] = environment_value

    return keys


def save_keys(keys):
    # Store only supported providers.
    safe_keys = {
        provider: str(keys.get(provider, "")).strip()
        for provider in PROVIDERS
    }

    temporary_path = KEYS_PATH.with_suffix(".tmp")

    with open(temporary_path, "w", encoding="utf-8") as file:
        json.dump(
            safe_keys,
            file,
            ensure_ascii=False,
            indent=2,
        )

    temporary_path.replace(KEYS_PATH)

    # Restrict permissions where supported.
    try:
        os.chmod(KEYS_PATH, 0o600)
    except OSError:
        pass


def get_api_key(provider):
    if provider not in PROVIDERS:
        raise ValueError("مزود الخدمة غير معروف.")

    keys = load_keys()
    api_key = keys.get(provider, "").strip()

    if not api_key:
        raise ValueError(
            f"أضف مفتاح API الخاص بـ {PROVIDERS[provider]['name']} أولًا."
        )

    return api_key


# =========================================================
# HTTP HELPERS
# =========================================================

def request_json(
    method,
    url,
    *,
    headers=None,
    params=None,
    body=None,
    timeout=60,
):
    try:
        response = requests.request(
            method=method,
            url=url,
            headers=headers,
            params=params,
            json=body,
            timeout=timeout,
        )

    except requests.Timeout:
        raise RuntimeError("انتهت مهلة الاتصال بمزود الذكاء الاصطناعي.")

    except requests.RequestException as error:
        raise RuntimeError(f"تعذر الاتصال بالمزود: {error}")

    if not response.ok:
        message = response.text[:1500]

        try:
            error_data = response.json()
            message = json.dumps(
                error_data,
                ensure_ascii=False,
            )[:1500]
        except ValueError:
            pass

        raise RuntimeError(
            f"خطأ من المزود ({response.status_code}): {message}"
        )

    try:
        return response.json()

    except ValueError:
        raise RuntimeError("أعاد المزود استجابة غير صالحة.")

# GEMINI MODELS
# =========================================================

def get_gemini_models(api_key):
    url = (
        "https://generativelanguage.googleapis.com/"
        "v1beta/models"
    )

    data = request_json(
        "GET",
        url,
        params={"key": api_key},
        timeout=30,
    )

    models = []

    for model in data.get("models", []):
        name = model.get("name", "").replace("models/", "")

        supported_methods = model.get(
            "supportedGenerationMethods",
            [],
        )

        if (
            name in GEMINI_FREE_ALLOWLIST
            and "generateContent" in supported_methods
        ):
            models.append({
                "id": name,
                "name": name,
            })

    return sorted(models, key=lambda item: item["name"])


# =========================================================
# OPENROUTER MODELS
# =========================================================

def get_openrouter_models(api_key):
    url = "https://openrouter.ai/api/v1/models"

    data = request_json(
        "GET",
        url,
        headers={
            "Authorization": f"Bearer {api_key}",
        },
        timeout=30,
    )

    models = []

    for model in data.get("data", []):
        model_id = model.get("id", "")

        pricing = model.get("pricing", {})

        if not model_id:
            continue

        # Explicit free model IDs.
        is_free_id = model_id.endswith(":free")

        # Check whether the provider explicitly reports zero pricing.
        is_zero_priced = False

        try:
            prompt_price = float(pricing.get("prompt", -1))
            completion_price = float(pricing.get("completion", -1))

            is_zero_priced = (
                prompt_price == 0
                and completion_price == 0
            )
        except (TypeError, ValueError):
            pass

        if is_free_id or is_zero_priced:
            models.append({
                "id": model_id,
                "name": model.get("name", model_id),
            })

    # Optional free router, when supported.
    # It is listed as a free route rather than a specific model.
    if not any(
        model["id"] == "openrouter/free"
        for model in models
    ):
        models.insert(0, {
            "id": "openrouter/free",
            "name": "OpenRouter Free Router",
        })

    models.sort(
        key=lambda item: (
            item["id"] != "openrouter/free",
            item["name"].lower(),
        )
    )

    return models

def fetch_models(provider, api_key):
    if provider == "gemini":
        return get_gemini_models(api_key)

    if provider == "openrouter":
        return get_openrouter_models(api_key)

    if provider == "groq":
        return get_groq_models(api_key)

    if provider == "deepseek":
        return get_deepseek_models(api_key)

    raise ValueError("مزود الخدمة غير معروف.")


def is_allowed_model(provider, model):
    if not isinstance(model, str) or not model:
        return False

    if provider == "gemini":
        return model in GEMINI_FREE_ALLOWLIST

    if provider == "openrouter":
        return (
            model == "openrouter/free"
            or model.endswith(":free")
        )

    if provider == "groq":
        return model in GROQ_FREE_ALLOWLIST

    if provider == "deepseek":
        return False

    return False

def call_gemini(api_key, model, messages):
    url = (
        "https://generativelanguage.googleapis.com/"
        f"v1beta/models/{model}:generateContent"
    )

    contents = []

    for message in messages:
        role = message.get("role")
        content = message.get("content", "")

        if role not in ("user", "assistant"):
            continue

        if not isinstance(content, str) or not content:
            continue

        contents.append({
            "role": "model" if role == "assistant" else "user",
            "parts": [
                {"text": content}
            ],
        })

    if not contents:
        raise RuntimeError("لا توجد رسائل لإرسالها إلى Gemini.")

    data = request_json(
        "POST",
        url,
        params={"key": api_key},
        body={
            "contents": contents,
            "generationConfig": {
                "temperature": 0.7,
            },
        },
        timeout=120,
    )

    candidates = data.get("candidates", [])

    if not candidates:
        raise RuntimeError(
            "لم يُرجع Gemini إجابة. تحقق من المفتاح والموديل."
        )

    parts = candidates[0].get(
        "content", {}
    ).get("parts", [])

    answer = "\n".join(
        part.get("text", "")
        for part in parts
        if part.get("text")
    )

    if not answer.strip():
        raise RuntimeError("أعاد Gemini إجابة فارغة.")

    return answer


def call_compatible_api(provider, api_key, model, messages):
    base_url = PROVIDERS[provider]["base_url"]

    url = base_url.rstrip("/") + "/chat/completions"

    headers = {
        "Authorization": f"Bearer {api_key}",
        "Content-Type": "application/json",
    }

    if provider == "openrouter":
        headers["HTTP-Referer"] = "http://localhost:5000"
        headers["X-Title"] = "Local AI Assistant"

    data = request_json(
        "POST",
        url,
        headers=headers,
        body={
            "model": model,
            "messages": messages,
            "temperature": 0.7,
        },
        timeout=120,
    )

    choices = data.get("choices", [])

    if not choices:
        raise RuntimeError("لم يُرجع المزود إجابة.")

    message = choices[0].get("message", {})
    content = message.get("content", "")

    if isinstance(content, list):
        content = "\n".join(
            item.get("text", "")
            for item in content
            if isinstance(item, dict)
        )

    if not isinstance(content, str) or not content.strip():
        raise RuntimeError("أعاد المزود إجابة فارغة.")

    return content


def call_model(provider, model, messages):
    if provider not in PROVIDERS:
        raise ValueError("مزود الخدمة غير معروف.")

    if not is_allowed_model(provider, model):
        raise ValueError(
            "هذا الموديل غير موجود في قائمة الموديلات المجانية المسموح بها."
        )

    api_key = get_api_key(provider)

    if provider == "gemini":
        return call_gemini(api_key, model, messages)

    if provider in ("openrouter", "groq"):
        return call_compatible_api(
            provider,
            api_key,
            model,
            messages,
        )

    if provider == "deepseek":
        raise RuntimeError(
            "لا توجد موديلات مجانية مؤكدة في وضع DeepSeek API المباشر. "
            "جرّب البحث عن موديل DeepSeek مجاني عبر OpenRouter."
        )

    raise ValueError("مزود الخدمة غير مدعوم.")

def create_conversation(title="محادثة جديدة"):
    now = datetime.now().isoformat(timespec="seconds")

    with get_db() as db:
        cursor = db.execute(
            """
            INSERT INTO conversations
            (title, created_at, updated_at)
            VALUES (?, ?, ?)
            """,
            (title, now, now),
        )

        db.commit()

        return cursor.lastrowid


def conversation_exists(conversation_id):
    with get_db() as db:
        row = db.execute(
            "SELECT id FROM conversations WHERE id = ?",
            (conversation_id,),
        ).fetchone()

    return row is not None


def save_message(
    conversation_id,
    role,
    content,
    provider=None,
    model=None,
):
    now = datetime.now().isoformat(timespec="seconds")

    with get_db() as db:
        db.execute(
            """
            INSERT INTO messages
            (
                conversation_id,
                role,
                content,
                provider,
                model,
                created_at
            )
            VALUES (?, ?, ?, ?, ?, ?)
            """,
            (
                conversation_id,
                role,
                content,
                provider,
                model,
                now,
            ),
        )

        db.execute(
            """
            UPDATE conversations
            SET updated_at = ?
            WHERE id = ?
            """,
            (now, conversation_id),
        )

        # Use the first user message as the conversation title.
        if role == "user":
            row = db.execute(
                """
                SELECT title FROM conversations
                WHERE id = ?
                """,
                (conversation_id,),
            ).fetchone()

            if row and row["title"] == "محادثة جديدة":
                title = content.strip().replace("\n", " ")

                if len(title) > 45:
                    title = title[:45] + "..."

                if title:
                    db.execute(
                        """
                        UPDATE conversations
                        SET title = ?
                        WHERE id = ?
                        """,
                        (title, conversation_id),
                    )

        db.commit()


def get_conversation_messages(conversation_id):
    with get_db() as db:
        rows = db.execute(
            """
            SELECT role, content, provider, model, created_at
            FROM messages
            WHERE conversation_id = ?
            ORDER BY id ASC
            """,
            (conversation_id,),
        ).fetchall()

    return [dict(row) for row in rows]

@app.route("/api/keys", methods=["POST"])
def api_save_keys():
    data = request.get_json(silent=True) or {}

    current_keys = load_keys()

    for provider in PROVIDERS:
        if provider not in data:
            continue

        value = data[provider]

        if not isinstance(value, str):
            return jsonify({
                "error": f"قيمة مفتاح {provider} غير صالحة."
            }), 400

        value = value.strip()

        # Empty input keeps the existing key.
        if value:
            current_keys[provider] = value

    try:
        save_keys(current_keys)

    except OSError as error:
        return jsonify({
            "error": f"تعذر حفظ المفاتيح: {error}"
        }), 500

    return jsonify({
        "success": True,
        "message": "تم حفظ المفاتيح.",
        "keys": {
            provider: bool(current_keys.get(provider))
            for provider in PROVIDERS
        },
    })

                "تحقق من حدود حسابك وإتاحة الموديل."

        elif provider == "groq":
            notice = (
                "تعرض القائمة موديلات مسموحًا بها في الكود؛ "

            "notice": notice,
        })

    except (RuntimeError, ValueError) as error:
        return jsonify({
            "error": str(error),
            "models": [],
        }), 400

@app.route("/chat", methods=["POST"])
def chat():
    data = request.get_json(silent=True) or {}

    message = data.get("message", "")
    provider = data.get("provider", "")
    model = data.get("model", "")
    conversation_id = data.get("conversation_id")

    if not isinstance(message, str) or not message.strip():
        return jsonify({
            "error": "اكتب رسالة أولًا."
        }), 400

    message = message.strip()

    if len(message) > 100000:
        return jsonify({
            "error": "الرسالة طويلة جدًا."
        }), 400

    if provider not in PROVIDERS:
        return jsonify({
            "error": "اختر مزود ذكاء اصطناعي صحيحًا."
        }), 400

    if not isinstance(model, str) or not model:
        return jsonify({
            "error": "اختر موديلًا أولًا."
        }), 400

    if conversation_id in ("", None):
        conversation_id = None

    try:
        if conversation_id is not None:
            conversation_id = int(conversation_id)

            if not conversation_exists(conversation_id):
                return jsonify({
                    "error": "المحادثة غير موجودة."
                }), 404

        else:
            conversation_id = create_conversation()

        # Build the context from the existing conversation.
        previous_messages = get_conversation_messages(
            conversation_id
        )

        # Store the user's message.
        save_message(
            conversation_id,
            "user",
            message,
            provider,
            model,
        )

        # Send the existing context plus the new message.
        api_messages = []

        for old_message in previous_messages:
            if old_message["role"] in ("user", "assistant"):
                api_messages.append({
                    "role": old_message["role"],
                    "content": old_message["content"],
                })

        api_messages.append({
            "role": "user",
            "content": message,
        })

        answer = call_model(
            provider,
            model,
            api_messages,
        )

        save_message(
            conversation_id,
            "assistant",
            answer,
            provider,
            model,
        )

        with get_db() as db:
            row = db.execute(
                """
                SELECT title
                FROM conversations
                WHERE id = ?
                """,
                (conversation_id,),
            ).fetchone()

        return jsonify({
            "success": True,
            "conversation_id": conversation_id,
            "title": row["title"] if row else "محادثة جديدة",
            "answer": answer,
            "provider": provider,
            "model": model,
        })

    except (RuntimeError, ValueError) as error:
        return jsonify({
            "error": str(error),
            "conversation_id": conversation_id,
        }), 400

    except Exception:
        app.logger.exception("Unexpected chat error")

        return jsonify({
            "error": "حدث خطأ غير متوقع. راجع نافذة تشغيل Flask."
        }), 500
