import sqlite3
from pathlib import Path

DB_PATH = Path(__file__).resolve().parent / "assistant.db"

if not DB_PATH.exists():
    print("Database file not found.")
    raise SystemExit(1)

confirm = input(
    "Delete ALL conversations and messages permanently? Type YES: "
)

if confirm != "YES":
    print("Cancelled. No data was deleted.")
    raise SystemExit(0)

with sqlite3.connect(DB_PATH) as db:
    db.execute("PRAGMA foreign_keys = ON")

    db.execute("DELETE FROM messages")
    db.execute("DELETE FROM conversations")

    # Reset auto-increment counters.
    db.execute(
        "DELETE FROM sqlite_sequence "
        "WHERE name IN ('messages', 'conversations')"
    )

    db.commit()

print("All conversations and messages have been deleted.")
print("API keys and application settings were preserved.")
